const MAX_FILE_BYTES = 1.5 * 1024 * 1024;

function escapeHtml(s) {
  return String(s ?? '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}
function formatBody(t) { return escapeHtml(t || '').replace(/\n/g, '<br>'); }
function formatSize(n) {
  if (n == null) return '';
  if (n < 1024) return n + ' B';
  if (n < 1024 * 1024) return (n / 1024).toFixed(1) + ' KB';
  return (n / (1024 * 1024)).toFixed(1) + ' MB';
}
function readFileAsMeta(file) {
  return new Promise((resolve) => {
    if (!file) return resolve(null);
    const meta = { name: file.name, type: file.type || 'application/octet-stream', size: file.size, dataUrl: null, tooLarge: false };
    if (file.size > MAX_FILE_BYTES) { meta.tooLarge = true; return resolve(meta); }
    const reader = new FileReader();
    reader.onload = () => { meta.dataUrl = reader.result; resolve(meta); };
    reader.onerror = () => resolve(meta);
    reader.readAsDataURL(file);
  });
}
async function readFilesFromInput(inputEl, multiple) {
  const files = inputEl?.files ? Array.from(inputEl.files) : [];
  if (!files.length) return multiple ? [] : null;
  if (!multiple) return readFileAsMeta(files[0]);
  const out = [];
  for (const f of files) out.push(await readFileAsMeta(f));
  return out;
}
function fileLinkHtml(doc) {
  if (!doc) return '<span class="text-secondary">—</span>';
  const label = `${escapeHtml(doc.name || 'File')} (${formatSize(doc.size)})`;
  if (doc.tooLarge) return `<span class="badge-soft badge-warn">${label} — too large (max 1.5MB)</span>`;
  if (doc.dataUrl) return `<a href="${doc.dataUrl}" target="_blank" rel="noopener" style="color:#93c5fd">${label}</a>`;
  return `<span class="badge-soft">${label}</span>`;
}

function recordPayment({ learnerId, courseId, amount, method, ref }) {
  const payments = getPayments();
  payments.unshift({
    id: uid(), learnerId, courseId, amount, method: method || 'card',
    ref: ref || ('EF-' + Math.random().toString(36).slice(2, 8).toUpperCase()),
    at: new Date().toISOString(), status: 'completed',
  });
  savePayments(payments);
  const list = getEnrolments();
  const row = list.find((e) => e.learnerId === learnerId && e.courseId === courseId);
  if (row) { row.paymentStatus = 'paid'; row.paidAt = new Date().toISOString(); saveEnrolments(list); }
  return true;
}
function issueCertificate(learnerId, courseId) {
  const existing = getCertificates().find((c) => c.learnerId === learnerId && c.courseId === courseId);
  if (existing) return existing;
  const course = getCourse(courseId);
  const learner = load(KEYS.users, []).find((u) => u.id === learnerId);
  const cert = {
    id: uid(), learnerId, courseId,
    learnerName: learner?.name || 'Learner',
    programmeTitle: course?.programme || course?.title || 'Programme',
    nqfLevel: course?.nqfLevel, credits: course?.credits, saqaId: course?.saqaId || '',
    issuedAt: new Date().toISOString(),
    certificateNo: 'EF-' + new Date().getFullYear() + '-' + Math.random().toString(36).slice(2, 8).toUpperCase(),
  };
  const list = getCertificates(); list.unshift(cert); saveCertificates(list);
  if (learner) pushMessage(learner.id, learner.gmail || learner.email, 'Certificate issued', 'Your certificate for ' + cert.programmeTitle + ' is ready. Certificate no: ' + cert.certificateNo);
  return cert;
}

function applyTheme(theme) {
  const t = theme === 'light' ? 'light' : 'dark';
  document.documentElement.setAttribute('data-theme', t);
  try { localStorage.setItem('ef_theme', t); } catch {}
  const meta = document.getElementById('meta-theme-color');
  if (meta) meta.setAttribute('content', t === 'light' ? '#f1f5f9' : '#0f172a');

    if (!icon) return;
    icon.className = t === 'light' ? 'bi bi-sun-fill' : 'bi bi-moon-stars-fill';
  });
}

function enforceOneProgrammeRule() {
  const enrolments = getEnrolments();
  const byLearner = {};
  enrolments.forEach((e) => {
    if (e.status !== 'approved') return;
    if (!byLearner[e.learnerId]) byLearner[e.learnerId] = [];
    byLearner[e.learnerId].push(e);
  });
  let changed = false;
  Object.values(byLearner).forEach((list) => {
    if (list.length <= 1) return;
    list.sort((a, b) => String(a.enrolledAt || '').localeCompare(String(b.enrolledAt || '')));
    for (let i = 1; i < list.length; i++) {
      list[i].status = 'inactive';
      list[i].inactiveReason = 'One programme at a time';
      changed = true;
    }
  });
  if (changed) saveEnrolments(enrolments);
}

function todayISODate() {
  const d = new Date();
  return d.getFullYear() + '-' + String(d.getMonth()+1).padStart(2,'0') + '-' + String(d.getDate()).padStart(2,'0');
}
function markAttendance(learnerId, courseId, sessionLabel) {
  const list = getAttendance();
  const day = todayISODate();
  if (list.find((a) => a.learnerId === learnerId && a.courseId === courseId && a.date === day))
    return { ok: false, message: 'Already marked present for today on this programme.' };
  list.unshift({ id: uid(), learnerId, courseId, date: day, session: sessionLabel || 'Remote session', markedAt: new Date().toISOString(), method: 'remote' });
  saveAttendance(list);
  return { ok: true };
}

function login(email, password) {
  const users = load(KEYS.users, []);
  const user = users.find((u) => u.email.toLowerCase() === email.toLowerCase() && u.password === password);
  if (!user) return { ok: false, message: 'Invalid email or password' };
  if (user.role === 'learner' && user.status === 'unverified') { setSession(user); return { ok: true, user, needsVerify: true }; }
  if (user.role === 'learner' && user.status === 'registered') { setSession(user); return { ok: true, user, needsApplication: true }; }
  if (user.role === 'learner' && (user.status || 'approved') === 'pending') { setSession(user); return { ok: true, user, pending: true }; }
  if (user.role === 'learner' && user.status === 'rejected') return { ok: false, message: 'Your application was not approved. Contact the training provider.' };
  setSession(user);
  return { ok: true, user };
}
function createLearnerAccount(p) {
  const users = load(KEYS.users, []);
  const em = (p.email || '').trim().toLowerCase();
  const name = (p.name || '').trim();
  if (!name) return { ok: false, message: 'Full name is required' };
  if (!em.includes('@')) return { ok: false, message: 'Enter a valid email (Gmail preferred)' };
  if ((p.password || '').length < 6) return { ok: false, message: 'Password must be at least 6 characters' };
  if (p.password !== p.password2) return { ok: false, message: 'Passwords do not match' };
  if (users.some((u) => u.email.toLowerCase() === em)) return { ok: false, message: 'Email already registered — sign in instead' };
  const code = generateVerifyCode();
  const user = { id: uid(), name, email: em, password: p.password, role: 'learner', gmail: em, phone: (p.phone || '').trim(), status: 'unverified', preferredProgrammeId: null, createdAt: new Date().toISOString(), verifyCode: code, emailVerified: false };
  users.push(user); save(KEYS.users, users);
  pushMessage(user.id, em, 'Welcome — verify your email', 'Your verification code was sent to ' + em + '.');
  setSession(user);
  return { ok: true, user, needsVerify: true, code };
}
function verifyLearnerEmail(userId, code) {
  const users = load(KEYS.users, []);
  const user = users.find((u) => u.id === userId);
  if (!user) return { ok: false, message: 'Account not found. Please sign in again.' };
  if (String(code || '').trim() !== String(user.verifyCode || '')) return { ok: false, message: 'Incorrect verification code.' };
  user.emailVerified = true; user.status = 'registered'; user.verifiedAt = new Date().toISOString(); delete user.verifyCode;
  save(KEYS.users, users);
  pushMessage(user.id, user.gmail || user.email, 'Email verified', 'Complete your application to request access.');
  setSession(user);
  return { ok: true, user, needsApplication: true };
}
function submitApplicationForUser(userId, p) {
  const users = load(KEYS.users, []);
  const user = users.find((u) => u.id === userId);
  if (!user) return { ok: false, message: 'Account not found. Please sign in again.' };
  if (!p.programmeId) return { ok: false, message: 'Please select a preferred programme' };
  if (!p.documents?.idCopy) return { ok: false, message: 'Please upload your ID / Smart ID' };
  if (!p.documents?.matric) return { ok: false, message: 'Please upload your Matric certificate' };
  if (p.documents.hasDisability && !p.documents.disabilityNote) return { ok: false, message: 'Doctor note is required when disability is selected' };
  user.status = 'pending'; user.preferredProgrammeId = p.programmeId; user.appliedAt = new Date().toISOString();
  save(KEYS.users, users);
  const apps = load(KEYS.applications, []).filter((a) => !(a.userId === userId && a.status === 'pending'));
  apps.unshift({ id: uid(), userId, name: user.name, email: user.email, phone: user.phone || '', programmeId: p.programmeId, notes: p.notes || '', documents: p.documents || null, status: 'pending', submittedAt: new Date().toISOString(), reviewNotes: '' });
  save(KEYS.applications, apps);
  pushMessage(user.id, user.gmail || user.email, 'Application received — awaiting approval', 'Your application and documents are under review.');
  setSession(user);
  return { ok: true, user, pending: true };
}
function approveAccountApplication(appId, approve, reviewNotes) {
  const applications = load(KEYS.applications, []);
  const app = applications.find((a) => a.id === appId);
  if (!app) return false;
  app.status = approve ? 'approved' : 'rejected'; app.reviewNotes = reviewNotes || ''; app.reviewedAt = new Date().toISOString();
  save(KEYS.applications, applications);
  const users = load(KEYS.users, []);
  const user = users.find((u) => u.id === app.userId);
  if (user) {
    user.status = approve ? 'approved' : 'rejected';
    save(KEYS.users, users);
    if (approve && user.preferredProgrammeId && !hasActiveProgramme(user.id)) {
      const list = getEnrolments().filter((e) => !(e.learnerId === user.id && e.courseId === user.preferredProgrammeId));
      list.push({ id: uid(), courseId: user.preferredProgrammeId, learnerId: user.id, status: 'approved', paymentStatus: 'unpaid', enrolledAt: new Date().toISOString() });
      saveEnrolments(list);
    }
    pushMessage(user.id, user.gmail || user.email, approve ? 'Application approved' : 'Application not approved', approve ? 'Approved. Pay your programme fee to unlock modules. One programme at a time.' : 'Not approved. ' + (reviewNotes || ''));
  }
  return true;
}

      if (k === 'programmes') { renderCatalog(); setNavActive('catalog'); }
      if (k === 'learners') renderLearners();
      if (k === 'pending') { renderApprovals(); setNavActive('approvals'); }
      if (k === 'grade') { renderGrading(); setNavActive('grade'); }
      if (k === 'attend') { renderAttendance(); setNavActive('attend'); }
      if (k === 'sessions') { renderLiveSessions(); setNavActive('sessions'); }
      if (k === 'payments') { renderPayments(); setNavActive('payments'); }
    });
  });
}
function renderLearners() {
  const users = load(KEYS.users, []).filter((u) => u.role === 'learner');
  const courses = getCourses();
  content().innerHTML = `<h2 class="page-title">Learners</h2><p class="page-subtitle">One active programme · payment &amp; certificate status</p>` + users.map((u) => {
    const st = u.status || 'approved';
    const enr = getApprovedEnrolments(u.id)[0];
    const prog = enr ? courses.find((c) => c.id === enr.courseId) : null;
    const pay = enr ? getEnrolmentPaymentStatus(enr) : null;
    return `<div class="list-card"><h3>${escapeHtml(u.name)}</h3><p class="text-secondary small">${escapeHtml(u.email)}</p><span class="badge-soft ${st==='approved'?'badge-ok':'badge-warn'}">${st}</span>${prog ? `<p class="text-secondary small mt-1">${escapeHtml(prog.title)} · ${pay}</p><button class="btn btn-sm btn-outline-light mt-1" data-complete="${enr.id}">Mark complete + certificate</button>` : '<p class="text-secondary small mt-1 mb-0">No active programme</p>'}</div>`;
  }).join('');
  content().querySelectorAll('[data-complete]').forEach((btn) => {
    btn.addEventListener('click', () => {
      if (!confirm('Mark complete and issue certificate?')) return;
      const list = getEnrolments();
      const row = list.find((x) => x.id === btn.getAttribute('data-complete'));
      if (row) {
        row.status = 'completed'; row.completedAt = new Date().toISOString();
        saveEnrolments(list);
        issueCertificate(row.learnerId, row.courseId);
      }
      renderLearners();
    });
  });
}
function renderApprovals() {
  if (!isAdmin(getSession())) { renderAttendance(); return; }
  const users = load(KEYS.users, []);
  const courses = getCourses();
  const pendingAccounts = load(KEYS.applications, []).filter((a) => a.status === 'pending');
  const pendingEnrol = getEnrolments().filter((e) => e.status === 'pending');
  let html = `<h2 class="page-title">Applications &amp; access</h2><p class="page-subtitle">Admin only · document review · one programme rule</p>`;
  if (pendingAccounts.length) {
    html += `<h3 class="h6 text-secondary">Account applications (${pendingAccounts.length})</h3>`;
    pendingAccounts.forEach((a) => {
      const prog = courses.find((c) => c.id === a.programmeId);
      const d = a.documents || {};
      html += `<div class="list-card" style="border-color:#eab308"><h3>${escapeHtml(a.name)}</h3><p>${escapeHtml(a.email)}</p><p class="text-secondary small">${escapeHtml(prog ? (prog.programme || prog.title) : '—')} · ${formatMoney(prog?.feeZAR)}</p><div class="content-block mb-2" style="font-size:.85rem"><p class="mb-1">ID: ${fileLinkHtml(d.idCopy)}</p><p class="mb-1">Matric: ${fileLinkHtml(d.matric)}</p></div><textarea class="form-control form-control-sm mb-2" id="acc-notes-${a.id}" rows="2"></textarea><button class="btn btn-sm btn-primary me-2" data-acc-approve="${a.id}">Approve</button><button class="btn btn-sm btn-outline-light" data-acc-decline="${a.id}">Decline</button></div>`;
    });
  }
  if (pendingEnrol.length) {
    html += `<h3 class="h6 text-secondary mt-3">Course requests (${pendingEnrol.length})</h3>`;
    pendingEnrol.forEach((e) => {
      const learner = users.find((u) => u.id === e.learnerId);
      const course = courses.find((c) => c.id === e.courseId);
      const blocked = hasActiveProgramme(e.learnerId);
      html += `<div class="list-card"><h3>${escapeHtml(learner?.name || 'Learner')}</h3><p>${escapeHtml(course?.title || '')}</p>${blocked ? '<p class="text-secondary small">Already has active programme.</p>' : ''}<button class="btn btn-sm btn-primary" data-approve="${e.id}" ${blocked?'disabled':''}>Approve</button><button class="btn btn-sm btn-outline-light" data-decline="${e.id}">Decline</button></div>`;
    });
  }
  if (!pendingAccounts.length && !pendingEnrol.length) html += '<div class="empty">No pending applications</div>';
  content().innerHTML = html;
  content().querySelectorAll('[data-acc-approve]').forEach((btn) => {
    btn.addEventListener('click', () => { approveAccountApplication(btn.getAttribute('data-acc-approve'), true, document.getElementById('acc-notes-' + btn.getAttribute('data-acc-approve'))?.value || ''); renderApprovals(); });
  });
  content().querySelectorAll('[data-acc-decline]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const id = btn.getAttribute('data-acc-decline');
      const notes = document.getElementById('acc-notes-' + id)?.value || '';
      if (!notes.trim()) { alert('Add a reason when declining.'); return; }
      approveAccountApplication(id, false, notes); renderApprovals();
    });
  });
  content().querySelectorAll('[data-approve]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const list = getEnrolments();
      const row = list.find((x) => x.id === btn.getAttribute('data-approve'));
      if (!row || hasActiveProgramme(row.learnerId)) return;
      row.status = 'approved'; row.paymentStatus = 'unpaid'; saveEnrolments(list); renderApprovals();
    });
  });
  content().querySelectorAll('[data-decline]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const list = getEnrolments();
      const row = list.find((x) => x.id === btn.getAttribute('data-decline'));
      if (row) { row.status = 'declined'; saveEnrolments(list); }
      renderApprovals();
    });
  });
}
