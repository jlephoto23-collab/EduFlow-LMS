const MAX_FILE_BYTES = 1.5 * 1024 * 1024;

function escapeHtml(s) {
  return String(s ?? '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}
function formatBody(t) {
  return escapeHtml(t || '').replace(/\n/g, '<br>');
}
function formatMoney(n) {
  return 'R' + Number(n || 0).toLocaleString('en-ZA');
}

function focusPageHeading() {
  const main = document.getElementById('content');
  if (!main) return;
  const heading = main.querySelector('h1, h2.page-title, .page-title, .course-hero h1, .welcome-banner h2');
  const target = heading || main;
  if (!target.hasAttribute('tabindex')) target.setAttribute('tabindex', '-1');
  try { target.focus({ preventScroll: false }); } catch (_) { try { target.focus(); } catch (__) {} }
}
function announce(message) {
  const el = document.getElementById('sr-status');
  if (!el || !message) return;
  el.textContent = '';
  window.setTimeout(() => { el.textContent = String(message); }, 50);
}
function pageNavBar(opts) {
  const backLabel = (opts && opts.backLabel) || 'Back';
  const nextLabel = opts && opts.nextLabel;
  let html = `<div class="page-nav d-flex align-items-center gap-2 mb-2">
    <button type="button" class="btn btn-sm btn-outline-light" id="btn-page-back" aria-label="${escapeHtml(backLabel)}">
      <i class="bi bi-arrow-left"></i> ${escapeHtml(backLabel)}
    </button>`;
  if (nextLabel) {
    html += `<button type="button" class="btn btn-sm btn-primary ms-auto" id="btn-page-next" aria-label="${escapeHtml(nextLabel)}">
      ${escapeHtml(nextLabel)} <i class="bi bi-arrow-right"></i>
    </button>`;
  }
  html += `</div>`;
  return html;
}
function wirePageNav(backFn, nextFn) {

  if (nextFn) document.getElementById('btn-page-next')?.addEventListener('click', (e) => { e.preventDefault(); nextFn(); });
}

function getEnrolmentPaymentStatus(enr) {
  if (!enr) return 'unpaid';
  if (enr.paymentStatus === 'paid' || enr.paymentStatus === 'pending' || enr.paymentStatus === 'unpaid') return enr.paymentStatus;
  return 'unpaid';
}
function recordPayment({ learnerId, courseId, amount, method, ref }) {
  const payments = getPayments();
  payments.unshift({ id: uid(), learnerId, courseId, amount, method: method || 'card', ref: ref || ('EF-' + Math.random().toString(36).slice(2, 8).toUpperCase()), at: new Date().toISOString(), status: 'completed' });
  savePayments(payments);
  const list = getEnrolments();
  const row = list.find((e) => e.learnerId === learnerId && e.courseId === courseId && e.status === 'approved');
  if (row) { row.paymentStatus = 'paid'; saveEnrolments(list); }
  if (typeof announce === 'function') announce('Payment recorded.');
}

function enforceOneProgrammeRule() {
  const enrolments = getEnrolments();
  const byLearner = {};
  enrolments.forEach((e) => {
    if (e.status !== 'approved') return;
    if (!byLearner[e.learnerId]) byLearner[e.learnerId] = [];
    byLearner[e.learnerId].push(e);
  });
  let changed = false;
  Object.values(byLearner).forEach((list) => {
    if (list.length <= 1) return;
    list.sort((a, b) => String(b.enrolledAt || '').localeCompare(String(a.enrolledAt || '')));
    list.slice(1).forEach((e) => { e.status = 'completed'; e.paymentStatus = e.paymentStatus || 'unpaid'; changed = true; });
  });
  if (changed) saveEnrolments(enrolments);
}

function todayISODate() { return new Date().toISOString().slice(0, 10); }
function markAttendance(learnerId, courseId) {
  const list = getAttendance();
  const day = todayISODate();
  if (list.some((a) => a.learnerId === learnerId && a.courseId === courseId && a.date === day)) return;
  list.unshift({ id: uid(), learnerId, courseId, date: day, at: new Date().toISOString() });

}

async function readFilesFromInput(input, multiple) {
  if (!input || !input.files || !input.files.length) return multiple ? [] : null;
  const files = multiple ? Array.from(input.files) : [input.files[0]];
  const out = [];
  for (const file of files) {
    if (file.size > MAX_FILE_BYTES) { out.push({ name: file.name, type: file.type, size: file.size, dataUrl: null, tooLarge: true }); continue; }
    const dataUrl = await new Promise((resolve, reject) => { const r = new FileReader(); r.onload = () => resolve(r.result); r.onerror = reject; r.readAsDataURL(file); });
    out.push({ name: file.name, type: file.type, size: file.size, dataUrl, tooLarge: false });
  }
  return multiple ? out : out[0];
}
function fileLinkHtml(f) {
  if (!f) return '';
  if (f.tooLarge || !f.dataUrl) return escapeHtml(f.name || 'file') + ' (unavailable)';
  return `<a href="${f.dataUrl}" target="_blank" rel="noopener">${escapeHtml(f.name || 'file')}</a>`;
}
function generateVerifyCode() { return String(Math.floor(100000 + Math.random() * 900000)); }
function createLearnerAccount({ name, email, phone, password, password2 }) {
  if (!name || !email || !password) return { ok: false, message: 'Complete all required fields.' };
  if (password.length < 6) return { ok: false, message: 'Password must be at least 6 characters.' };
  if (password !== password2) return { ok: false, message: 'Passwords do not match.' };
  const users = load(KEYS.users, []);
  if (users.some((u) => String(u.email).toLowerCase() === email.toLowerCase())) return { ok: false, message: 'Email already registered.' };
  const code = generateVerifyCode();
  const user = { id: uid(), name, email, phone, password, role: 'learner', gmail: email, status: 'unverified', verifyCode: code, emailVerified: false };
  users.push(user); save(KEYS.users, users); setSession(user);
  return { ok: true, user, code };
}
function verifyLearnerEmail(userId, code) {
  const users = load(KEYS.users, []);
  const user = users.find((u) => u.id === userId);
  if (!user) return { ok: false, message: 'Session expired.' };
  if (String(user.verifyCode) !== String(code).trim()) return { ok: false, message: 'Invalid code.' };
  user.emailVerified = true; user.status = 'verified'; user.verifyCode = null;
  save(KEYS.users, users); setSession(user);
  return { ok: true, user };
}
function submitApplicationForUser(userId, { programmeId, notes, documents }) {
  if (!programmeId) return { ok: false, message: 'Select a programme.' };
  if (!documents?.idCopy || !documents?.matric) return { ok: false, message: 'ID and Matric uploads are required.' };
  if (documents.hasDisability && !documents.disabilityNote) return { ok: false, message: 'Doctor note required for disability.' };
  if (hasActiveProgramme(userId)) return { ok: false, message: 'You already have an active programme.' };
  const list = getEnrolments().filter((e) => !(e.learnerId === userId && e.courseId === programmeId));
  list.push({ id: uid(), courseId: programmeId, learnerId: userId, status: 'pending', paymentStatus: 'unpaid', enrolledAt: new Date().toISOString(), notes, documents });
  saveEnrolments(list);
  const users = load(KEYS.users, []);
  const user = users.find((u) => u.id === userId);
  if (user) { user.status = 'pending'; save(KEYS.users, users); setSession(user); }
  return { ok: true };
}
function login(email, password) {
  const users = load(KEYS.users, []);
  const user = users.find((u) => String(u.email).toLowerCase() === String(email).toLowerCase() && u.password === password);
  if (!user) return { ok: false, message: 'Invalid email or password.' };
  setSession(user);
  if (user.role === 'learner') {
    if (user.status === 'unverified' || user.emailVerified === false) return { ok: true, needsVerify: true, user };
    const enr = getEnrolments().filter((e) => e.learnerId === user.id);
    if (!enr.length && user.status !== 'approved') return { ok: true, needsApplication: true, user };
    if (enr.every((e) => e.status === 'pending') || user.status === 'pending') return { ok: true, pending: true, user };
  }
  return { ok: true, user };
}

function toggleTheme() {
  const cur = document.documentElement.getAttribute('data-theme') === 'light' ? 'light' : 'dark';
  const next = cur === 'light' ? 'dark' : 'light';
  document.documentElement.setAttribute('data-theme', next);
  try { localStorage.setItem('ef_theme', next); } catch (_) {}
  wireThemeToggles();
}
function wireThemeToggles() {
  const theme = document.documentElement.getAttribute('data-theme') === 'light' ? 'light' : 'dark';
  ['theme-icon', 'theme-icon-login'].forEach((id) => {
    const el = document.getElementById(id);
    if (el) el.className = theme === 'light' ? 'bi bi-moon-stars-fill' : 'bi bi-sun-fill';
  });
  document.getElementById('btn-theme-toggle')?.addEventListener('click', toggleTheme);
  document.getElementById('btn-theme-login')?.addEventListener('click', toggleTheme);
}

function renderCourseDetail(courseId) {
  const c = getCourse(courseId);
  if (!c) return renderCatalog();
  const s = getSession();
  const enr = getEnrolments().find(e => e.learnerId === s.id && e.courseId === courseId);
  const pay = getEnrolmentPaymentStatus(enr);
  const lessons = c.lessons || [];
  let html = pageNavBar({ backLabel: 'Back to Programmes' }) + `<div class="course-hero"><h1>${escapeHtml(c.saqaId || '')} — ${escapeHtml(c.programme || c.title)}</h1><p class="mb-0">NQF ${c.nqfLevel} · ${c.credits || ''} credits · ${formatMoney(c.feeZAR)}</p></div>`;
  if (isStaff(s)) {
    html += `<p class="text-secondary">Staff view of programme modules.</p>` + lessons.map((l, i) => `<div class="list-card list-link" data-lesson="${l.id}" data-idx="${i}"><h3>${escapeHtml(l.title)}</h3><span class="badge-soft">${escapeHtml(l.type || 'content')}</span></div>`).join('');
  } else if (!enr || enr.status !== 'approved') {
    html += `<div class="content-block"><p>You are not enrolled in this programme yet.</p><button type="button" class="btn btn-primary" id="btn-apply-here">Apply with documents</button></div>`;
  } else if (pay !== 'paid') {
    html += `<div class="content-block"><p>Payment outstanding. Complete payment to unlock modules.</p><button type="button" class="btn btn-primary" id="btn-pay-now">Go to Fees & Payment</button></div>`;
  } else {
    html += lessons.map((l, i) => `<div class="list-card list-link" data-lesson="${l.id}" data-idx="${i}"><h3>${escapeHtml(l.title)}</h3><span class="badge-soft">${escapeHtml(l.type || 'content')}</span></div>`).join('');
  }
  content().innerHTML = html;
  wirePageNav(() => { renderCatalog(); setNavActive('courses'); });
  document.getElementById('btn-pay-now')?.addEventListener('click', () => { renderPayments(); setNavActive('payments'); });
  document.getElementById('btn-apply-here')?.addEventListener('click', () => renderApplyInApp(courseId));
  content().querySelectorAll('[data-lesson]').forEach(el => {
    el.addEventListener('click', () => openLesson(courseId, el.getAttribute('data-lesson'), Number(el.getAttribute('data-idx') || 0)));
  });
  focusPageHeading();
}
function renderApplyInApp(courseId) {
  const s = getSession();
  if (!s) return showLogin();
  const c = getCourse(courseId);
  const courses = getCourses();
  content().innerHTML = pageNavBar({ backLabel: 'Back to programme' }) +
    `<h2 class="page-title">Apply with documents</h2>
    <p class="page-subtitle">Submit ID and Matric for ${escapeHtml(c?.programme || 'this programme')}</p>
    <form id="inapp-apply-form" class="content-block">
      <label class="form-label">Programme</label>
      <select id="inapp-programme" class="form-control mb-2" required>
        ${courses.map(x => `<option value="${x.id}" ${x.id === courseId ? 'selected' : ''}>${escapeHtml(x.programme || x.title)}</option>`).join('')}
      </select>
      <label class="form-label">ID / Smart ID *</label>
      <input type="file" id="inapp-id" class="form-control mb-2" accept=".pdf,.jpg,.jpeg,.png,image/*,application/pdf" required>
      <label class="form-label">Matric certificate *</label>
      <input type="file" id="inapp-matric" class="form-control mb-2" accept=".pdf,.jpg,.jpeg,.png,image/*,application/pdf" required>
      <label class="form-label">Other certificates (optional)</label>
      <input type="file" id="inapp-other" class="form-control mb-2" accept=".pdf,.jpg,.jpeg,.png,image/*,application/pdf" multiple>
      <div class="form-check mb-2">
        <input class="form-check-input" type="checkbox" id="inapp-disability">
        <label class="form-check-label text-secondary small" for="inapp-disability">Disability (doctor note required)</label>
      </div>
      <label class="form-label">Doctor note</label>
      <input type="file" id="inapp-note" class="form-control mb-2" accept=".pdf,.jpg,.jpeg,.png,image/*,application/pdf">
      <label class="form-label">Notes</label>
      <textarea id="inapp-notes" class="form-control mb-3" rows="2"></textarea>
      <div id="inapp-error" class="alert alert-danger d-none py-2" role="alert"></div>
      <button type="submit" class="btn btn-primary w-100">Submit application</button>
    </form>`;
  wirePageNav(() => renderCourseDetail(courseId));
  document.getElementById('inapp-apply-form')?.addEventListener('submit', async (e) => {
    e.preventDefault();
    const err = document.getElementById('inapp-error');
    const programmeId = document.getElementById('inapp-programme')?.value;
    try {
      const idCopy = await readFilesFromInput(document.getElementById('inapp-id'), false);
      const matric = await readFilesFromInput(document.getElementById('inapp-matric'), false);
      const other = await readFilesFromInput(document.getElementById('inapp-other'), true);
      const hasDisability = !!document.getElementById('inapp-disability')?.checked;
      const disabilityNote = hasDisability ? await readFilesFromInput(document.getElementById('inapp-note'), false) : null;
      const notes = document.getElementById('inapp-notes')?.value || '';
      const result = submitApplicationForUser(s.id, { programmeId, notes, documents: { idCopy, matric, other, hasDisability, disabilityNote } });
      if (!result.ok) {
        if (err) { err.textContent = result.message; err.classList.remove('d-none'); }
        announce(result.message || 'Could not submit');
        return;
      }
      announce('Application submitted. Awaiting admin approval.');
      content().innerHTML = pageNavBar({ backLabel: 'Back to Home' }) +
        `<h2 class="page-title">Application submitted</h2>
        <div class="content-block"><p>Your application for <strong>${escapeHtml(getCourse(programmeId)?.programme || '')}</strong> was received.</p>
        <p class="mb-0 text-secondary">An admin will review your documents.</p></div>
        <button type="button" class="btn btn-primary w-100 mt-3" id="btn-after-apply">Back to Home</button>`;
      wirePageNav(goHome);
      document.getElementById('btn-after-apply')?.addEventListener('click', goHome);
    } catch (ex) {
      if (err) { err.textContent = 'Could not read files. Try smaller PDFs or images.'; err.classList.remove('d-none'); }
    }
  });
  focusPageHeading();
}
function openLesson(courseId, lessonId, idx) {
  const c = getCourse(courseId);
  const lessons = c?.lessons || [];
  const lesson = lessons.find(l => l.id === lessonId) || lessons[idx];
  if (!lesson) return renderCourseDetail(courseId);
  const i = typeof idx === 'number' ? idx : lessons.findIndex(l => l.id === lessonId);
  const hasPrev = i > 0;
  const hasNext = i < lessons.length - 1;
  content().innerHTML = pageNavBar({ backLabel: 'Back to course', nextLabel: hasNext ? 'Next lesson' : null }) +
    `<h2 class="page-title">${escapeHtml(lesson.title)}</h2>
    <p class="text-secondary small mb-2">Lesson ${i + 1} of ${lessons.length}</p>
    <div class="content-block">${formatBody(lesson.body || 'Content for this lesson.')}</div>
    <div class="d-flex gap-2 mt-3">
      ${hasPrev ? '<button type="button" class="btn btn-outline-light" id="btn-prev-lesson"><i class="bi bi-arrow-left"></i> Previous</button>' : ''}
      <button type="button" class="btn btn-primary ${hasPrev ? 'ms-auto' : 'w-100'}" id="btn-mark-done">Mark as Done</button>
    </div>`;
  wirePageNav(() => renderCourseDetail(courseId), hasNext ? () => openLesson(courseId, lessons[i + 1].id, i + 1) : null);
  document.getElementById('btn-prev-lesson')?.addEventListener('click', () => openLesson(courseId, lessons[i - 1].id, i - 1));
  document.getElementById('btn-mark-done')?.addEventListener('click', () => {
    announce('Marked as done');
    if (hasNext) openLesson(courseId, lessons[i + 1].id, i + 1);
    else renderCourseDetail(courseId);
  });
  focusPageHeading();
}
function renderPayments() {
  const s = getSession();
  const approved = getApprovedEnrolments(s.id);
  const courses = getCourses();
  content().innerHTML = pageNavBar({ backLabel: 'Back to Home' }) + `<h2 class="page-title">Fees & payments</h2>` +
    (approved.length === 0 ? '<div class="empty">No approved programmes</div>' : approved.map(e => {
      const c = courses.find(x => x.id === e.courseId);
      const pay = getEnrolmentPaymentStatus(e);
      return `<div class="list-card"><h3>${escapeHtml(c?.programme || '')}</h3><p>${formatMoney(c?.feeZAR)} · <span class="badge-soft ${pay==='paid'?'badge-ok':'badge-warn'}">${pay}</span></p>${pay !== 'paid' ? `<button type="button" class="btn btn-sm btn-primary" data-pay="${e.courseId}">Pay now</button>` : ''}</div>`;
    }).join(''));
  wirePageNav(goHome);
  content().querySelectorAll('[data-pay]').forEach(btn => {
    btn.addEventListener('click', () => {
      const courseId = btn.getAttribute('data-pay');
      const c = getCourse(courseId);
      recordPayment({ learnerId: s.id, courseId, amount: c?.feeZAR || 0, method: 'card' });
      renderPayments();
    });
  });
  focusPageHeading();
}
