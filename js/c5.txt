/** Strict view isolation + dynamic verify email + demo fills */

function showView(viewId) {
  document.querySelectorAll('.view').forEach((el) => {
    const active = el.id === viewId;
    el.classList.toggle('is-active', active);
    if (active) {
      el.removeAttribute('hidden');
      el.removeAttribute('inert');
      el.setAttribute('aria-hidden', 'false');
    } else {
      el.setAttribute('hidden', '');
      el.setAttribute('inert', '');
      el.setAttribute('aria-hidden', 'true');
    }
  });
  const signedIn = viewId === 'view-workspace';
  document.getElementById('btn-logout')?.classList.toggle('d-none', !signedIn);
  document.getElementById('user-chip')?.classList.toggle('d-none', !signedIn);
  try { window.scrollTo({ top: 0, behavior: 'smooth' }); } catch (_) { window.scrollTo(0, 0); }

  // Move document title focus for a11y: one logical screen
  const active = document.getElementById(viewId);
  const heading = active?.querySelector('h1, h2.view-title, .view-title');
  if (heading) {
    if (!heading.hasAttribute('tabindex')) heading.setAttribute('tabindex', '-1');
    try { heading.focus({ preventScroll: true }); } catch (_) {}
  }
}

function goToVerifyStep(emailValue, user) {
  const cleanEmail = String(emailValue || user?.email || getSession()?.email || '').trim() || 'your email';
  const target = document.getElementById('verifyEmailDisplay');
  if (target) target.textContent = cleanEmail;
  const legacy = document.getElementById('verify-email-label');
  if (legacy) legacy.textContent = cleanEmail;
  const demo = document.getElementById('verify-demo-code');
  if (demo) {
    if (user?.verifyCode) {
      demo.classList.remove('d-none');
      demo.textContent = 'Demo code: ' + user.verifyCode;
    } else {
      demo.classList.add('d-none');
      demo.textContent = '';
    }
  }
  showView('view-step-2');
}

function showLogin() {
  try { setSession(null); } catch (_) {}
  // Kill any splash leftovers
  document.getElementById('splash')?.classList.add('hide');
  showView('view-signin');
}

function showRegisterPanel() {
  showView('view-step-1');
}

function showVerifyPanel(user) {
  goToVerifyStep(user?.email, user);
}

function showApplicationStep(user, preferredCourseId) {
  const sel = document.getElementById('apply-programme');
  if (sel && typeof getCourses === 'function') {
    sel.innerHTML = getCourses().map((c) =>
      '<option value="' + c.id + '"' + (preferredCourseId && c.id === preferredCourseId ? ' selected' : '') + '>' +
      escapeHtml(c.programme || c.title) + '</option>'
    ).join('');
  }
  showView('view-step-3');
}

function showApp() {
  try { if (typeof ensureDemoEnrolments === 'function') ensureDemoEnrolments(); } catch (_) {}
  document.getElementById('splash')?.classList.add('hide');
  let s = getSession();
  if (!s) return showLogin();
  const fresh = load(KEYS.users, []).find((u) => u.id === s.id || String(u.email).toLowerCase() === String(s.email || '').toLowerCase());
  if (fresh) { setSession(fresh); s = fresh; }
  showView('view-workspace');
  const chip = document.getElementById('user-chip');
  if (chip) chip.textContent = (s.name || s.email) + ' · ' + (s.role || '');
  if (typeof renderBottomNav === 'function') renderBottomNav(s);
  if (typeof isStaff === 'function' && isStaff(s)) {
    if (typeof renderAdminHome === 'function') renderAdminHome();
  } else if (typeof renderLearnerHome === 'function') {
    renderLearnerHome();
  }
  if (typeof setNavActive === 'function') setNavActive('home');
  if (typeof focusPageHeading === 'function') focusPageHeading();
}

function wireDemoFills() {
  document.querySelectorAll('.demo-fill').forEach((btn) => {
    // avoid double-binding
    if (btn.dataset.wired === '1') return;
    btn.dataset.wired = '1';
    btn.addEventListener('click', () => {
      const email = btn.getAttribute('data-email') || '';
      const pass = btn.getAttribute('data-pass') || '';
      const role = btn.getAttribute('data-role') || btn.textContent || 'Demo';
      const e = document.getElementById('email');
      const p = document.getElementById('password');
      if (e) e.value = email;
      if (p) p.value = pass;
      document.querySelectorAll('.demo-fill').forEach((b) => b.classList.remove('active'));
      btn.classList.add('active');
      const status = document.getElementById('demo-fill-status');
      if (status) status.textContent = '✓ ' + role + ' credentials filled — press Sign in';
      if (typeof announce === 'function') announce(role + ' credentials filled');
      e?.focus();
    });
  });
}

function wireOnboardingForms() {
  const once = (id, evt, fn) => {
    const el = document.getElementById(id);
    if (!el || el.dataset['wired' + evt] === '1') return;
    el.dataset['wired' + evt] = '1';
    el.addEventListener(evt, fn);
  };

  once('btn-register', 'click', (e) => { e.preventDefault(); showRegisterPanel(); });
  once('btn-show-login-from-reg', 'click', (e) => { e.preventDefault(); showLogin(); });
  once('btn-show-login-from-verify', 'click', (e) => { e.preventDefault(); showLogin(); });
  once('btn-show-login', 'click', (e) => {
    e.preventDefault();
    if (typeof logout === 'function') logout();
    else showLogin();
  });

  once('register-form', 'submit', (e) => {
    e.preventDefault();
    const err = document.getElementById('reg-error');
    const email = document.getElementById('reg-email')?.value?.trim() || '';
    const password = document.getElementById('reg-password')?.value || '';
    const password2 = document.getElementById('reg-password2')?.value || '';
    if (!email) {
      if (err) { err.textContent = 'Enter your email address.'; err.classList.remove('d-none'); }
      return;
    }
    if (password !== password2) {
      if (err) { err.textContent = 'Passwords do not match.'; err.classList.remove('d-none'); }
      return;
    }
    if (password.length < 6) {
      if (err) { err.textContent = 'Password must be at least 6 characters.'; err.classList.remove('d-none'); }
      return;
    }
    const result = createLearnerAccount({
      name: document.getElementById('reg-name')?.value?.trim(),
      email,
      phone: document.getElementById('reg-phone')?.value?.trim(),
      password,
      password2,
    });
    if (!result.ok) {
      if (err) { err.textContent = result.message; err.classList.remove('d-none'); }
      return;
    }
    if (err) err.classList.add('d-none');
    // Always pass the typed email so Step 2 is never a blank/generic string when known
    goToVerifyStep(email, result.user);
  });

  once('verify-form', 'submit', (e) => {
    e.preventDefault();
    const s = getSession();
    const err = document.getElementById('verify-error');
    if (!s) {
      if (err) { err.textContent = 'Session expired. Sign in again.'; err.classList.remove('d-none'); }
      return;
    }
    const code = document.getElementById('verify-code')?.value?.trim();
    const result = verifyLearnerEmail(s.id, code);
    if (!result.ok) {
      if (err) { err.textContent = result.message; err.classList.remove('d-none'); }
      return;
    }
    if (err) err.classList.add('d-none');
    showApplicationStep(result.user);
  });

  once('btn-resend-code', 'click', (e) => {
    e.preventDefault();
    const s = getSession();
    if (!s) return showLogin();
    const users = load(KEYS.users, []);
    const user = users.find((u) => u.id === s.id);
    if (user) {
      user.verifyCode = generateVerifyCode();
      save(KEYS.users, users);
      setSession(user);
      goToVerifyStep(user.email, user);
      if (typeof announce === 'function') announce('New demo code generated');
    }
  });

  const applyForm = document.getElementById('apply-form');
  if (applyForm && applyForm.dataset.wiredSize !== '1') {
    applyForm.dataset.wiredSize = '1';
    applyForm.addEventListener('submit', (e) => {
      const err = document.getElementById('apply-error');
      const max = 1.5 * 1024 * 1024;
      const files = ['apply-id', 'apply-matric', 'apply-note']
        .map((id) => document.getElementById(id)?.files?.[0])
        .filter(Boolean);
      for (const f of files) {
        if (f.size > max) {
          e.preventDefault();
          e.stopImmediatePropagation();
          if (err) {
            err.textContent = '"' + f.name + '" is larger than 1.5 MB. Please upload a smaller file.';
            err.classList.remove('d-none');
          }
          return;
        }
      }
    }, true);
  }

  wireDemoFills();
}

// Initial isolation before any other boot UI runs
(function initViews() {
  document.querySelectorAll('.view').forEach((el) => {
    if (!el.classList.contains('is-active')) {
      el.setAttribute('hidden', '');
      el.setAttribute('inert', '');
      el.setAttribute('aria-hidden', 'true');
    }
  });
  const run = () => { try { wireOnboardingForms(); } catch (err) { console.error(err); } };
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run);
  else run();
  setTimeout(run, 50);
  setTimeout(run, 250);
  // After modules define login handlers, force the sign-in view once
  setTimeout(() => {
    if (!getSession()) showView('view-signin');
  }, 300);
})();
