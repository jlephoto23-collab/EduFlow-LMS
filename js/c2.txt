/** Force demo learner Aisha onto approved+paid Project Management */
function ensureDemoEnrolments() {
  const users = load(KEYS.users, []);
  let usersChanged = false;
  const need = [
    { id: 'u_admin', name: 'Admin Trainer', email: 'admin@eduflow.test', password: 'admin123', role: 'admin', status: 'approved' },
    { id: 'u_facil', name: 'Facilitator Demo', email: 'facilitator@eduflow.test', password: 'facil123', role: 'facilitator', status: 'approved' },
    { id: 'u_l1', name: 'Aisha Learner', email: 'learner1@gmail.com', password: 'learner123', role: 'learner', status: 'approved' },
    { id: 'u_pend', name: 'Thabo Applicant', email: 'thabo.apply@gmail.com', password: 'pending123', role: 'learner', status: 'pending' },
  ];
  need.forEach((n) => {
    let u = users.find((x) => String(x.email).toLowerCase() === n.email.toLowerCase());
    if (!u) {
      users.push({ ...n, gmail: n.email, emailVerified: true });
      usersChanged = true;
    } else {
      // Keep stable id for Aisha so enrolments match session
      if (n.email === 'learner1@gmail.com') {
        if (u.id !== 'u_l1') { u.id = 'u_l1'; usersChanged = true; }
        if (u.status !== 'approved') { u.status = 'approved'; usersChanged = true; }
        if (u.password !== 'learner123') { u.password = 'learner123'; usersChanged = true; }
        u.emailVerified = true;
        u.role = 'learner';
      }
      if (n.email === 'admin@eduflow.test' && u.role !== 'admin') { u.role = 'admin'; u.status = 'approved'; usersChanged = true; }
    }
  });
  if (usersChanged) save(KEYS.users, users);

  const aisha = users.find((u) => String(u.email).toLowerCase() === 'learner1@gmail.com') || users.find((u) => u.id === 'u_l1');
  if (!aisha) return;

  const list = getEnrolments();
  let enrChanged = false;
  // Migrate any enrolment rows that pointed at old Aisha ids
  list.forEach((e) => {
    const learner = users.find((u) => u.id === e.learnerId);
    if (!learner && e.learnerId && e.courseId === 'c_pm4') {
      // orphaned row — retarget to Aisha if it looks like demo
    }
  });
  let row = list.find((e) => e.courseId === 'c_pm4' && (e.learnerId === aisha.id || e.learnerId === 'u_l1'));
  if (!row) {
    list.push({
      id: 'enr1',
      learnerId: aisha.id,
      courseId: 'c_pm4',
      status: 'approved',
      paymentStatus: 'paid',
      enrolledAt: new Date().toISOString(),
    });
    enrChanged = true;
  } else {
    if (row.learnerId !== aisha.id) { row.learnerId = aisha.id; enrChanged = true; }
    if (row.status !== 'approved') { row.status = 'approved'; enrChanged = true; }
    if (row.paymentStatus !== 'paid') { row.paymentStatus = 'paid'; enrChanged = true; }
  }
  if (enrChanged) saveEnrolments(list);

  // Sync session if logged in as Aisha with wrong id
  const sess = getSession();
  if (sess && String(sess.email).toLowerCase() === 'learner1@gmail.com' && sess.id !== aisha.id) {
    setSession(aisha);
  }
}

/** Login: approved enrolment wins over stale pending user.status */
function login(email, password) {
  try { ensureDemoEnrolments(); } catch (_) {}
  const users = load(KEYS.users, []);
  const user = users.find((u) => String(u.email).toLowerCase() === String(email).toLowerCase() && u.password === password);
  if (!user) return { ok: false, message: 'Invalid email or password.' };
  setSession(user);
  if (user.role === 'learner') {
    if (user.status === 'unverified' || user.emailVerified === false) return { ok: true, needsVerify: true, user };
    const enr = getEnrolments().filter((e) => e.learnerId === user.id);
    const hasApproved = enr.some((e) => e.status === 'approved');
    if (hasApproved) return { ok: true, user };
    if (!enr.length && user.status !== 'approved') return { ok: true, needsApplication: true, user };
    if (enr.length && enr.every((e) => e.status === 'pending')) return { ok: true, pending: true, user };
    if (user.status === 'pending' && !hasApproved) return { ok: true, pending: true, user };
  }
  return { ok: true, user };
}

/** Course detail: resolve enrolment by learner id; repair Aisha on the fly */
function renderCourseDetail(courseId) {
  try { ensureDemoEnrolments(); } catch (_) {}
  const c = getCourse(courseId);
  if (!c) return renderCatalog();
  let s = getSession();
  if (!s) return showLogin();
  // Refresh session user from store (id may have been repaired)
  const fresh = load(KEYS.users, []).find((u) => u.id === s.id || String(u.email).toLowerCase() === String(s.email || '').toLowerCase());
  if (fresh) { s = fresh; setSession(fresh); }

  let enr = getEnrolments().find((e) => e.learnerId === s.id && e.courseId === courseId);
  // Fallback: demo Aisha + PM
  if (!enr && String(s.email).toLowerCase() === 'learner1@gmail.com' && courseId === 'c_pm4') {
    ensureDemoEnrolments();
    enr = getEnrolments().find((e) => e.learnerId === s.id && e.courseId === courseId)
      || getEnrolments().find((e) => e.learnerId === 'u_l1' && e.courseId === courseId);
  }
  const pay = getEnrolmentPaymentStatus(enr);
  const lessons = c.lessons || [];
  let html = pageNavBar({ backLabel: 'Back to Programmes' }) +
    `<div class="course-hero"><h1>${escapeHtml(c.saqaId || '')} — ${escapeHtml(c.programme || c.title)}</h1>
    <p class="mb-0">NQF ${c.nqfLevel} · ${c.credits || ''} credits · ${formatMoney(c.feeZAR)}</p></div>`;

  if (isStaff(s)) {
    html += `<p class="text-secondary">Staff view of programme modules.</p>` +
      lessons.map((l, i) => `<div class="list-card list-link" data-lesson="${l.id}" data-idx="${i}"><h3>${escapeHtml(l.title)}</h3><span class="badge-soft">${escapeHtml(l.type || 'content')}</span></div>`).join('');
  } else if (!enr || enr.status !== 'approved') {
    html += `<div class="content-block"><p>You are not enrolled in this programme yet.</p>
      <p class="small text-secondary">If you already applied, wait for admin approval — or use the demo learner account (learner1@gmail.com).</p>
      <button type="button" class="btn btn-primary" id="btn-apply-here">Apply with documents</button></div>`;
  } else if (pay !== 'paid') {
    html += `<div class="content-block"><p>Payment outstanding. Complete payment to unlock modules.</p>
      <button type="button" class="btn btn-primary" id="btn-pay-now">Go to Fees & Payment</button></div>`;
  } else {
    html += lessons.map((l, i) => {
      const done = typeof isLessonDone === 'function' ? isLessonDone(s.id, courseId, l.id) : false;
      return `<div class="list-card list-link" data-lesson="${l.id}" data-idx="${i}">
        <h3>${escapeHtml(l.title)}</h3>
        <span class="badge-soft">${escapeHtml(l.type || 'content')}</span>
        ${done ? ' <span class="badge-soft badge-ok">Done</span>' : ''}
      </div>`;
    }).join('');
  }
  content().innerHTML = html;
  wirePageNav(() => { renderCatalog(); setNavActive('courses'); });
  document.getElementById('btn-pay-now')?.addEventListener('click', () => { renderPayments(); setNavActive('payments'); });
  document.getElementById('btn-apply-here')?.addEventListener('click', () => {
    if (typeof renderApplyInApp === 'function') renderApplyInApp(courseId);
    else if (typeof showApplicationStep === 'function') showApplicationStep(s, courseId);
  });
  content().querySelectorAll('[data-lesson]').forEach((el) => {
    el.addEventListener('click', () => openLesson(courseId, el.getAttribute('data-lesson'), Number(el.getAttribute('data-idx') || 0)));
  });
  focusPageHeading();
}

function showApp() {
  try { ensureDemoEnrolments(); } catch (_) {}
  const s = getSession();
  if (!s) return showLogin();
  // Re-bind session to store user (stable ids)
  const fresh = load(KEYS.users, []).find((u) => u.id === s.id || String(u.email).toLowerCase() === String(s.email || '').toLowerCase());
  if (fresh) setSession(fresh);
  const sess = getSession();
  document.getElementById('login-view')?.classList.add('d-none');
  document.getElementById('app-view')?.classList.remove('d-none');
  const chip = document.getElementById('user-chip');
  if (chip) chip.textContent = (sess.name || sess.email) + ' · ' + (sess.role || '');
  renderBottomNav(sess);
  if (isStaff(sess)) renderAdminHome(); else renderLearnerHome();
  setNavActive('home');
  focusPageHeading();
}

function renderGrading() {
  const s = getSession();
  if (!isStaff(s)) {
    content().innerHTML = pageNavBar({ backLabel: 'Back to Home' }) + '<h2 class="page-title">Grading</h2><div class="empty">Staff only.</div>';
    wirePageNav(goHome);
    focusPageHeading();
    return;
  }
  const RUBRIC = [
    { id: 'clarity', label: 'Task clarity & structure' },
    { id: 'content', label: 'Content accuracy' },
    { id: 'evidence', label: 'Evidence / examples' },
    { id: 'presentation', label: 'Presentation & language' },
  ];
  const passScore = 6;
  const users = load(KEYS.users, []);
  const courses = getCourses();
  const list = getSubmissionsList().slice().sort((a, b) => String(b.at || '').localeCompare(String(a.at || '')));
  const pending = list.filter((x) => x.status === 'submitted');
  const graded = list.filter((x) => x.status === 'graded');
  let html = pageNavBar({ backLabel: 'Back to Home' }) +
    '<h2 class="page-title">Grading</h2>' +
    '<p class="page-subtitle">Rubric (0–2 each) · Competent / Not yet competent · Max 2 attempts</p>' +
    '<div class="content-block mb-3"><p class="small mb-0">Score each criterion. Total out of 8. <strong>6+</strong> suggests Competent — you can still override.</p></div>';
  if (!list.length) {
    html += '<div class="empty">No submissions yet.<br><span class="text-secondary small">Learner assignment uploads appear here with the rubric.</span></div>';
  } else {
    if (pending.length) {
      html += '<h3 class="h6 mb-2">To grade (' + pending.length + ')</h3>';
      pending.forEach((sub) => {
        const u = users.find((x) => x.id === sub.learnerId);
        const c = courses.find((x) => x.id === sub.courseId);
        html += '<div class="list-card">' +
          '<h3>' + escapeHtml(u?.name || 'Learner') + '</h3>' +
          '<p class="text-secondary small mb-1">' + escapeHtml(c?.programme || '') + ' · Attempt ' + (sub.attempt || 1) + ' · ' + escapeHtml(sub.fileName || 'file') + '</p>' +
          '<p class="small mb-2">' + escapeHtml(sub.notes || '') + '</p>' +
          (sub.fileDataUrl && !sub.tooLarge ? '<p class="small mb-2"><a href="' + sub.fileDataUrl + '" target="_blank" rel="noopener">Open submitted file</a></p>' : '') +
          '<div class="rubric-box" id="rubric-' + sub.id + '">' +
          '<h4>Assessment rubric</h4>' +
          RUBRIC.map((r) => '<div class="rubric-row"><span>' + escapeHtml(r.label) + '</span>' +
            '<select data-rubric="' + sub.id + '" data-crit="' + r.id + '" aria-label="' + escapeHtml(r.label) + '">' +
            '<option value="0">0</option><option value="1">1</option><option value="2" selected>2</option></select></div>').join('') +
          '<div class="rubric-total"><span>Total</span><span id="rubric-total-' + sub.id + '">8 / 8</span></div>' +
          '<p class="small text-secondary mb-0 mt-1" id="rubric-hint-' + sub.id + '">Suggested: Competent</p></div>' +
          '<label class="form-label">Feedback to learner</label>' +
          '<input type="text" class="form-control mb-2" id="fb-' + sub.id + '" placeholder="Short, actionable feedback">' +
          '<div class="d-flex gap-2 flex-wrap">' +
          '<button type="button" class="btn btn-sm btn-primary" data-grade="' + sub.id + '" data-result="Competent">Mark Competent</button>' +
          '<button type="button" class="btn btn-sm btn-outline-light" data-grade="' + sub.id + '" data-result="Not Yet Competent">Not yet competent</button>' +
          '</div></div>';
      });
    }
    if (graded.length) {
      html += '<h3 class="h6 mt-3 mb-2">Recently graded</h3>';
      graded.slice(0, 8).forEach((sub) => {
        const u = users.find((x) => x.id === sub.learnerId);
        const scores = sub.rubricScores || {};
        const total = Object.values(scores).reduce((a, b) => a + Number(b || 0), 0);
        html += '<div class="list-card"><h3>' + escapeHtml(u?.name || 'Learner') +
          ' <span class="badge-soft ' + (sub.grade === 'Competent' ? 'badge-ok' : 'badge-warn') + '">' + escapeHtml(sub.grade || '') + '</span></h3>' +
          '<p class="small text-secondary mb-0">Attempt ' + (sub.attempt || 1) +
          (Object.keys(scores).length ? ' · Rubric ' + total + '/8' : '') +
          ' · ' + escapeHtml(sub.feedback || '') + '</p></div>';
      });
    }
  }
  content().innerHTML = html;
  wirePageNav(goHome);
  function refreshRubric(subId) {
    let total = 0;
    content().querySelectorAll('select[data-rubric="' + subId + '"]').forEach((sel) => { total += Number(sel.value || 0); });
    const el = document.getElementById('rubric-total-' + subId);
    if (el) el.textContent = total + ' / 8';
    const hint = document.getElementById('rubric-hint-' + subId);
    if (hint) hint.textContent = total >= passScore ? 'Suggested: Competent' : 'Suggested: Not yet competent';
  }
  content().querySelectorAll('select[data-rubric]').forEach((sel) => {
    sel.addEventListener('change', () => refreshRubric(sel.getAttribute('data-rubric')));
  });
  content().querySelectorAll('[data-grade]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const id = btn.getAttribute('data-grade');
      const result = btn.getAttribute('data-result');
      const fb = document.getElementById('fb-' + id)?.value || '';
      const rubricScores = {};
      let total = 0;
      content().querySelectorAll('select[data-rubric="' + id + '"]').forEach((sel) => {
        const crit = sel.getAttribute('data-crit');
        const v = Number(sel.value || 0);
        rubricScores[crit] = v;
        total += v;
      });
      const all = getSubmissionsList();
      const row = all.find((x) => x.id === id);
      if (!row) return;
      row.status = 'graded';
      row.grade = result;
      row.feedback = fb;
      row.rubricScores = rubricScores;
      row.rubricTotal = total;
      row.gradedAt = new Date().toISOString();
      row.gradedBy = s.id;
      saveSubmissionsList(all);
      announce(result === 'Competent' ? 'Marked Competent' : 'Marked Not yet competent');
      renderGrading();
    });
  });
  focusPageHeading();
}

(function patchLearnerHomeProgress() {
  const orig = renderLearnerHome;
  if (typeof orig !== 'function') return;
  renderLearnerHome = function () {
    try { ensureDemoEnrolments(); } catch (_) {}
    orig();
    content().querySelectorAll('[data-open]').forEach((card) => {
      const bar = card.querySelector('div[style*="height:8px"]');
      if (bar && bar.parentElement) {
        const fill = bar.querySelector('div');
        const pct = fill && fill.style.width ? fill.style.width : '0%';
        const wrap = document.createElement('div');
        wrap.className = 'progress-track';
        wrap.innerHTML = '<div class="progress-fill" style="width:' + pct + '"></div>';
        bar.replaceWith(wrap);
      }
    });
  };
})();
