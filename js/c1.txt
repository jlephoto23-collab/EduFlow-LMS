function pageNavHtml({ backLabel, nextLabel, showNext, center }) {
  const back = backLabel || 'Back';
  const next = nextLabel || 'Next';
  return `<div class="page-nav" id="page-nav-top">
    <button type="button" class="btn btn-sm btn-outline-light" id="btn-page-back"><i class="bi bi-arrow-left"></i> ${escapeHtml(back)}</button>
    <span class="page-nav-center">${center ? escapeHtml(center) : ''}</span>
    ${showNext ? `<button type="button" class="btn btn-sm btn-primary" id="btn-page-next">${escapeHtml(next)} <i class="bi bi-arrow-right"></i></button>` : `<span style="min-width:6.5rem"></span>`}
  </div>`;
}
function pageNavBottomHtml({ backLabel, nextLabel, showNext }) {
  const back = backLabel || 'Back';
  const next = nextLabel || 'Next';
  return `<div class="page-nav page-nav-sticky" id="page-nav-bottom">
    <button type="button" class="btn btn-sm btn-outline-light" id="btn-page-back-b"><i class="bi bi-arrow-left"></i> ${escapeHtml(back)}</button>
    ${showNext ? `<button type="button" class="btn btn-sm btn-primary" id="btn-page-next-b">${escapeHtml(next)} <i class="bi bi-arrow-right"></i></button>` : `<span></span>`}
  </div>`;
}
function wirePageNav(onBack, onNext) {
  const goBack = () => { if (typeof onBack === 'function') onBack(); };
  const goNext = () => { if (typeof onNext === 'function') onNext(); };
  document.getElementById('btn-page-back')?.addEventListener('click', goBack);
  document.getElementById('btn-page-back-b')?.addEventListener('click', goBack);
  document.getElementById('btn-page-next')?.addEventListener('click', goNext);
  document.getElementById('btn-page-next-b')?.addEventListener('click', goNext);
}
function getCourseLessonList(courseId) {
  const c = getCourse(courseId);
  return (c && c.lessons) ? c.lessons.slice() : [];
}
function openAdjacentLesson(courseId, lessonId, dir) {
  const list = getCourseLessonList(courseId);
  const idx = list.findIndex((l) => l.id === lessonId);
  if (idx < 0) return renderCourseDetail(courseId);
  const next = list[idx + dir];
  if (!next) return renderCourseDetail(courseId);
  const s = getSession();
  openLesson(courseId, next, isStaff(s));
}

function renderCatalog() {
  const s = getSession();
  const courses = getCourses();
  const enrolments = getEnrolments();
  content().innerHTML = `<h2 class="page-title">${isStaff(s) ? 'Programmes' : 'Course catalogue'}</h2><p class="text-secondary mb-3">One active programme at a time.</p>` +
    courses.map((c) => {
      const enr = enrolments.find((e) => e.learnerId === s.id && e.courseId === c.id);
      const status = enr ? enr.status : null;
      return `<div class="list-card"><h3>${escapeHtml(c.title)}</h3><p class="text-secondary small mb-2">${escapeHtml(c.description || '')}</p><div class="meta mb-2"><span class="badge-soft">NQF ${c.nqfLevel}</span><span class="badge-soft">${formatMoney(c.feeZAR)}</span></div>${isStaff(s) || status === 'approved' ? `<button class="btn btn-sm btn-primary" data-open="${c.id}">Open modules</button>` : status === 'pending' ? `<span class="badge-soft badge-warn">Application pending</span>` : (s.role === 'learner' && hasActiveProgramme(s.id) ? `<span class="badge-soft">Finish current programme first</span>` : `<button class="btn btn-sm btn-primary" data-apply="${c.id}">Apply for this programme</button>`)}</div>`;
    }).join('');
  content().querySelectorAll('[data-open]').forEach((btn) => btn.addEventListener('click', () => renderCourseDetail(btn.getAttribute('data-open'))));
  content().querySelectorAll('[data-apply]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const courseId = btn.getAttribute('data-apply');
      if (hasActiveProgramme(s.id)) { alert('Complete your current programme first.'); return; }
      const list = getEnrolments().filter((e) => !(e.learnerId === s.id && e.courseId === courseId));
      list.push({ id: uid(), courseId, learnerId: s.id, status: 'pending', paymentStatus: 'unpaid', enrolledAt: new Date().toISOString() });
      saveEnrolments(list);
      alert('Application submitted.');
      renderCatalog();
    });
  });
}

function getLessonStatus(courseId, lesson, learnerId) {
  if (!learnerId) return { label: 'View', cls: '', done: false };
  if (lesson.type === 'guide' || lesson.type === 'content') {
    const prog = load(KEYS.progress, []).find((p) => p.learnerId === learnerId && p.courseId === courseId);
    const done = !!(prog && (prog.completedLessonIds || []).includes(lesson.id));
    return done ? { label: 'Done', cls: 'status-done', done: true } : { label: 'To do', cls: 'status-todo', done: false };
  }
  if (lesson.type === 'quiz') {
    const attempts = load(KEYS.attempts, []).filter((a) => a.learnerId === learnerId && a.lessonId === lesson.id);
    if (!attempts.length) return { label: 'Attempt 1 of 2', cls: 'status-todo', done: false };
    const best = attempts.reduce((a, b) => (a.score >= b.score ? a : b));
    if (best.passed) return { label: 'Competent', cls: 'status-done', done: true };
    if (attempts.length >= 2) return { label: 'Not Yet Competent', cls: 'status-warn', done: false };
    return { label: 'Attempt 2 of 2', cls: 'status-warn', done: false };
  }
  if (lesson.type === 'assignment') {
    const subs = getSubmissions().filter((x) => x.learnerId === learnerId && x.assignmentId === lesson.assignmentId);
    const sub = subs[subs.length - 1];
    if (!sub) return { label: 'Attempt 1 of 2', cls: 'status-todo', done: false };
    const attemptNo = sub.attemptNo || subs.length;
    if (sub.status === 'submitted') return { label: 'Submitted (' + attemptNo + '/2)', cls: 'status-warn', done: false };
    if (sub.status === 'graded' && sub.passed) return { label: 'Competent', cls: 'status-done', done: true };
    if (sub.status === 'graded') return attemptNo >= 2 ? { label: 'Not Yet Competent', cls: 'status-warn', done: false } : { label: 'Attempt 2 of 2', cls: 'status-warn', done: false };
  }
  return { label: 'To do', cls: 'status-todo', done: false };
}

function getDocFile(entry) {
  if (!entry || entry === true) return null;
  return entry.file || null;
}
const DOC_REQUIRES_UPLOAD = { id: true, matric: true, cv: true };
function isDocDone(entry, docId) {
  if (!entry) return false;
  if (entry === true) return !DOC_REQUIRES_UPLOAD[docId];
  if (!entry.done) return false;
  if (DOC_REQUIRES_UPLOAD[docId] && !entry.file) return false;
  return true;
}

    const enr = getEnrolments().find((e) => e.learnerId === s.id && e.courseId === courseId && e.status === 'approved');

        () => { renderLearnerHome(); setNavActive('home'); },
        () => { renderPayments(); setNavActive('payments'); }

    }

  const lessons = c.lessons || [];

  if (s && s.id) {

      const e = savedDocs[docId];
      if (!e) return;

    { id: 'policy', title: 'Learner Policy Document', icon: 'bi-file-earmark-text', done: isDocDone(savedDocs.policy, 'policy') },
    { id: 'id', title: 'ID Copy', icon: 'bi-person-vcard', done: isDocDone(savedDocs.id, 'id') },
    { id: 'matric', title: 'Matric / Secondary certificate', icon: 'bi-award', done: isDocDone(savedDocs.matric, 'matric') },
    { id: 'cv', title: 'Curriculum Vitae', icon: 'bi-file-person', done: isDocDone(savedDocs.cv, 'cv') },

  <div class="course-hero"><h1>${progCode}</h1>
    <p class="mb-0 small opacity-75">NQF Level ${c.nqfLevel} · ${c.credits} credits · ${formatMoney(c.feeZAR)}</p></div>
  <div class="course-tabs mb-3" id="course-tabs">
    <button type="button" class="course-tab active" data-tab="outline">Section outline</button>
    <button type="button" class="course-tab" data-tab="grades">Grades</button>
    <button type="button" class="course-tab" data-tab="competencies">Competencies</button>
    ${staff ? '<button type="button" class="course-tab" data-tab="participants">Participants</button>' : ''}
  </div><div id="course-tab-body"></div>`;
  const renderOutline = () => {

function openCourseDocument(courseId, docId) {
  const titles = { ann: 'Announcements', policy: 'Learner Policy', id: 'ID Copy', matric: 'Matric certificate', cv: 'Curriculum Vitae' };
  const s = getSession();
  const key = s ? ('ef_docs_' + s.id + '_' + courseId) : null;
  const store = key ? load(key, {}) : {};
  const entry = store[docId];
  const already = isDocDone(entry, docId);
  const existingFile = getDocFile(entry);
  const needsFile = !!DOC_REQUIRES_UPLOAD[docId];
  const canUpload = needsFile || docId === 'policy';
  const docOrder = ['ann', 'policy', 'id', 'matric', 'cv'];
  const dIdx = docOrder.indexOf(docId);
  const hasNextDoc = dIdx >= 0 && dIdx < docOrder.length - 1;
  let fileBlock = '';
  if (canUpload) {
    fileBlock = `<div class="content-block mb-3"><p class="mb-2"><strong>${needsFile ? 'Required upload' : 'Optional upload'}</strong></p>
      ${existingFile ? `<p class="mb-2 small">Current: ${fileLinkHtml(existingFile)}</p>` : '<p class="text-secondary small mb-2">No file yet.</p>'}
      <input type="file" id="doc-upload" class="form-control mb-2" accept=".pdf,.jpg,.jpeg,.png,image/*,application/pdf,.doc,.docx">
      <button type="button" class="btn btn-sm btn-outline-light" id="btn-doc-upload"><i class="bi bi-upload me-1"></i>${existingFile ? 'Replace' : 'Upload'} file</button>
      <span id="doc-upload-status" class="text-secondary small ms-2"></span></div>`;
  }
  content().innerHTML = `${pageNavHtml({ backLabel: 'Course', nextLabel: 'Next doc', showNext: hasNextDoc, center: (dIdx + 1) + ' / ' + docOrder.length })}
    <h2 class="page-title">${escapeHtml(titles[docId] || 'Document')}</h2>
    <div class="content-block mb-3">${formatBody(docId === 'cv' ? 'Upload your current CV (PDF or Word).' : docId === 'id' ? 'Upload a clear ID copy.' : docId === 'matric' ? 'Upload Matric or equivalent.' : 'Read and confirm.')}</div>${fileBlock}
    ${already ? `<span class="badge-soft badge-ok">Completed</span><button class="btn btn-outline-light w-100 mt-3" id="btn-back-doc2">Back to course</button>` : `
    <div class="form-check mb-3"><input class="form-check-input" type="checkbox" id="doc-ack"><label class="form-check-label" for="doc-ack">${needsFile ? 'I uploaded the correct document' : 'I have read this document'}</label></div>
    <div id="doc-error" class="alert alert-danger d-none py-2"></div>
    <button class="btn btn-primary w-100" id="btn-doc-done" disabled>Mark as Done</button>`}
    ${pageNavBottomHtml({ backLabel: 'Course', nextLabel: 'Next doc', showNext: hasNextDoc })}`;
  wirePageNav(
    () => renderCourseDetail(courseId),
    () => { if (hasNextDoc) openCourseDocument(courseId, docOrder[dIdx + 1]); }
  );
  document.getElementById('btn-back-doc2')?.addEventListener('click', () => renderCourseDetail(courseId));
  document.getElementById('doc-ack')?.addEventListener('change', () => {
    const btn = document.getElementById('btn-doc-done');
    const ack = document.getElementById('doc-ack');
    const st = key ? load(key, {}) : {};
    if (btn) btn.disabled = !(ack?.checked && (!needsFile || !!getDocFile(st[docId])));
  });
  document.getElementById('btn-doc-upload')?.addEventListener('click', async () => {
    const input = document.getElementById('doc-upload');
    const status = document.getElementById('doc-upload-status');
    if (!input?.files?.length) { if (status) status.textContent = 'Choose a file first.'; return; }
    const fileRec = await readFilesFromInput(input, false);
    if (!fileRec || !s || !key) { if (status) status.textContent = 'Upload failed.'; return; }
    const st = load(key, {});
    st[docId] = { done: false, file: fileRec, uploadedAt: new Date().toISOString() };
    save(key, st);
    openCourseDocument(courseId, docId);
  });
  document.getElementById('btn-doc-done')?.addEventListener('click', () => {
    if (!document.getElementById('doc-ack')?.checked) return;
    if (!s || !key) return;
    const st = load(key, {});
    if (needsFile && !getDocFile(st[docId])) {
      const err = document.getElementById('doc-error');
      if (err) { err.textContent = 'Upload required first.'; err.classList.remove('d-none'); }
      return;
    }
    st[docId] = { done: true, file: getDocFile(st[docId]), completedAt: new Date().toISOString() };
    save(key, st);
    renderCourseDetail(courseId);
  });
}

function renderModuleSection(courseId, mod) {
  const c = getCourse(courseId);
  const s = getSession();
  const learnerId = isStaff(s) ? null : s.id;
  content().innerHTML = `${pageNavHtml({ backLabel: 'Course', nextLabel: 'First activity', showNext: mod.lessons.length > 0, center: mod.title })}
  <h2 class="page-title">${escapeHtml(mod.title)}</h2>
  <div class="apex-card">${mod.lessons.map((l) => {
    const st = getLessonStatus(courseId, l, learnerId);
    const icon = l.type === 'quiz' ? 'bi-pencil-square' : l.type === 'assignment' ? 'bi-upload' : 'bi-file-earmark-text';
    return `<div class="activity-row clickable" data-lesson="${l.id}" role="button"><div class="activity-left"><i class="bi ${icon}"></i><span>${escapeHtml(l.title)}</span></div><span class="status-pill ${st.cls}">${st.label}</span></div>`;
  }).join('')}</div>
  ${pageNavBottomHtml({ backLabel: 'Course', nextLabel: 'First activity', showNext: mod.lessons.length > 0 })}`;
  wirePageNav(
    () => renderCourseDetail(courseId),
    () => { if (mod.lessons[0]) openLesson(courseId, mod.lessons[0], isStaff(s)); }
  );
  content().querySelectorAll('[data-lesson]').forEach((el) => {
    el.addEventListener('click', () => {
      const lesson = (c.lessons || []).find((x) => x.id === el.getAttribute('data-lesson'));
      if (lesson) openLesson(courseId, lesson, isStaff(s));
    });
  });
}

function openLesson(courseId, lesson, isTrainer) {
  if (lesson.type === 'quiz') {
    const quiz = lesson.quiz || { questions: [], passMark: 70 };
    const qs = quiz.questions || [];
    const s = getSession();
    const prevAttempts = load(KEYS.attempts, []).filter((a) => a.learnerId === s.id && a.lessonId === lesson.id);
    const attemptNo = prevAttempts.length + 1;
    const best = prevAttempts.length ? prevAttempts.reduce((a, b) => (a.score >= b.score ? a : b)) : null;
    const listQ = getCourseLessonList(courseId);
    const idxQ = listQ.findIndex((l) => l.id === lesson.id);
    const hasNextQ = idxQ >= 0 && idxQ < listQ.length - 1;
    if (best && best.passed) {
      content().innerHTML = `${pageNavHtml({ backLabel: 'Back', nextLabel: 'Next', showNext: hasNextQ, center: 'Result' })}
        <h2 class="page-title">${escapeHtml(lesson.title)}</h2>
        <div class="content-block"><span class="badge-soft badge-ok">Competent</span><p class="mt-2 mb-0">Score: ${best.score}%</p></div>
        ${pageNavBottomHtml({ backLabel: 'Back', nextLabel: 'Next', showNext: hasNextQ })}`;
      wirePageNav(() => renderCourseDetail(courseId), () => openAdjacentLesson(courseId, lesson.id, 1)); return;
    }
    if (prevAttempts.length >= 2) {
      content().innerHTML = `${pageNavHtml({ backLabel: 'Back', nextLabel: 'Next', showNext: hasNextQ, center: 'Result' })}
        <h2 class="page-title">${escapeHtml(lesson.title)}</h2>
        <div class="content-block"><span class="badge-soft badge-warn">Not Yet Competent</span><p class="mt-2 mb-0">Best: ${best ? best.score : 0}% after 2 attempts.</p></div>
        ${pageNavBottomHtml({ backLabel: 'Back', nextLabel: 'Next', showNext: hasNextQ })}`;
      wirePageNav(() => renderCourseDetail(courseId), () => openAdjacentLesson(courseId, lesson.id, 1)); return;
    }
    content().innerHTML = `${pageNavHtml({ backLabel: 'Back', nextLabel: 'Next activity', showNext: hasNextQ, center: (idxQ + 1) + ' / ' + listQ.length })}
      <h2 class="page-title">${escapeHtml(lesson.title)}</h2>
      <p class="text-secondary mb-2">${qs.length} questions · pass ${quiz.passMark}% · <strong>Attempt ${attemptNo} of 2</strong></p>` +
      qs.map((q, qi) => `<div class="content-block mb-2"><p class="mb-2"><strong>${qi + 1}. ${escapeHtml(q.q)}</strong></p>${(q.options || []).map((opt, oi) => `<label class="d-block mb-1"><input type="radio" name="q${qi}" value="${oi}"> ${escapeHtml(opt)}</label>`).join('')}</div>`).join('') +
      `<div id="quiz-result" class="d-none content-block mb-2"></div><button class="btn btn-primary w-100" id="btn-submit-quiz">Submit attempt ${attemptNo}</button>
      ${pageNavBottomHtml({ backLabel: 'Back', nextLabel: 'Next activity', showNext: hasNextQ })}`;
    wirePageNav(() => renderCourseDetail(courseId), () => openAdjacentLesson(courseId, lesson.id, 1));
    document.getElementById('btn-submit-quiz')?.addEventListener('click', () => {
      let correct = 0;
      qs.forEach((q, qi) => { const sel = document.querySelector(`input[name="q${qi}"]:checked`); if (sel && Number(sel.value) === Number(q.correctIndex)) correct++; });
      const score = qs.length ? Math.round((correct / qs.length) * 100) : 0;
      const passed = score >= (quiz.passMark || 70);
      const attempts = load(KEYS.attempts, []);
      attempts.push({ id: uid(), learnerId: getSession().id, courseId, lessonId: lesson.id, score, passed, attemptNo, outcome: passed ? 'Competent' : 'Not Yet Competent', at: new Date().toISOString() });
      save(KEYS.attempts, attempts);
      const box = document.getElementById('quiz-result');
      if (box) { box.classList.remove('d-none'); box.innerHTML = `<span class="badge-soft ${passed ? 'badge-ok' : 'badge-warn'}">${passed ? 'Competent' : 'Not Yet Competent'} — ${score}%</span>`; }
      const btn = document.getElementById('btn-submit-quiz'); if (btn) btn.disabled = true;
    });
    return;
  }
  if (lesson.type === 'assignment') return openAssignmentLesson(courseId, lesson, isTrainer);
  const listC = getCourseLessonList(courseId);
  const idxC = listC.findIndex((l) => l.id === lesson.id);
  const hasNextC = idxC >= 0 && idxC < listC.length - 1;
  const hasPrevC = idxC > 0;
  content().innerHTML = `${pageNavHtml({ backLabel: hasPrevC ? 'Previous' : 'Module', nextLabel: 'Next', showNext: hasNextC, center: (idxC + 1) + ' / ' + listC.length })}
    <h2 class="page-title">${escapeHtml(lesson.title)}</h2><div class="content-block mb-3">${formatBody(lesson.body)}</div>
    <div class="form-check mb-3"><input class="form-check-input" type="checkbox" id="lesson-ack"><label class="form-check-label" for="lesson-ack">I have read this section</label></div>
    <button class="btn btn-primary w-100" id="btn-mark-done" disabled>Mark as Done</button>
    ${pageNavBottomHtml({ backLabel: hasPrevC ? 'Previous' : 'Back', nextLabel: 'Next', showNext: hasNextC })}`;
  wirePageNav(
    () => { if (hasPrevC) openAdjacentLesson(courseId, lesson.id, -1); else renderCourseDetail(courseId); },
    () => openAdjacentLesson(courseId, lesson.id, 1)
  );
  const ackL = document.getElementById('lesson-ack');
  const btnL = document.getElementById('btn-mark-done');
  ackL?.addEventListener('change', () => { if (btnL) btnL.disabled = !ackL.checked; });
  btnL?.addEventListener('click', () => {
    if (!ackL?.checked) return;
    const s = getSession();
    if (s.role === 'learner') {
      const all = load(KEYS.progress, []);
      let p = all.find((x) => x.learnerId === s.id && x.courseId === courseId);
      if (!p) { p = { learnerId: s.id, courseId, completedLessonIds: [], percent: 0 }; all.push(p); }
      if (!p.completedLessonIds.includes(lesson.id)) p.completedLessonIds.push(lesson.id);
      p.percent = Math.round((p.completedLessonIds.length / ((getCourse(courseId)?.lessons || []).length || 1)) * 100);
      save(KEYS.progress, all);
    }
    renderCourseDetail(courseId);
  });
}

function openAssignmentLesson(courseId, lesson, isTrainer) {
  const s = getSession();
  const mine = getSubmissions().filter((x) => x.learnerId === s.id && x.assignmentId === lesson.assignmentId).pop();
  const listA = getCourseLessonList(courseId);
  const idxA = listA.findIndex((l) => l.id === lesson.id);
  const hasNextA = idxA >= 0 && idxA < listA.length - 1;
  if (isTrainer) {
    content().innerHTML = `${pageNavHtml({ backLabel: 'Back', showNext: false, center: 'Assignment' })}
      <h2 class="page-title">${escapeHtml(lesson.title)}</h2><div class="content-block">${formatBody(lesson.body)}</div>`;
    wirePageNav(() => renderCourseDetail(courseId), null); return;
  }
  if (mine && (mine.status === 'submitted' || mine.status === 'graded')) {
    const files = [mine.attachments?.knowledge, mine.attachments?.practical].filter(Boolean);
    content().innerHTML = `${pageNavHtml({ backLabel: 'Back', nextLabel: 'Next', showNext: hasNextA, center: 'Submission' })}
    <h2 class="page-title">${escapeHtml(lesson.title)}</h2>
    <div class="content-block mb-3">${formatBody(mine.text || '—')}</div>
    <table class="status-table"><tr><th>Attempt</th><td>${mine.attemptNo || 1} of 2</td></tr>
    <tr><th>Outcome</th><td>${mine.status === 'graded' ? (mine.passed ? 'Competent' : 'Not Yet Competent') : 'Submitted for grading'}</td></tr>
    <tr><th>Files</th><td>${files.map(fileLinkHtml).join('<br>') || '—'}</td></tr></table>
    ${mine.status === 'graded' && !mine.passed && (mine.attemptNo || 1) < 2 ? '<button class="btn btn-primary w-100 mt-3" id="btn-update-submission">Add a new attempt (Remediation)</button>' : mine.status === 'submitted' ? '<button class="btn btn-outline-light w-100 mt-3" id="btn-update-submission">Update submission</button>' : ''}
    ${pageNavBottomHtml({ backLabel: 'Back', nextLabel: 'Next', showNext: hasNextA })}`;
    wirePageNav(() => renderCourseDetail(courseId), () => openAdjacentLesson(courseId, lesson.id, 1));
    document.getElementById('btn-update-submission')?.addEventListener('click', () => showAssignmentSubmitForm(courseId, lesson, mine));
    return;
  }
  showAssignmentSubmitForm(courseId, lesson, null);
}

function showAssignmentSubmitForm(courseId, lesson, previous) {
  const s = getSession();
  const subs = getSubmissions().filter((x) => x.learnerId === s.id && x.assignmentId === lesson.assignmentId);
  const gradedCount = subs.filter((x) => x.status === 'graded').length;
  const attemptNo = previous && previous.status === 'submitted' ? (previous.attemptNo || 1) : gradedCount + 1;
  const listA = getCourseLessonList(courseId);
  const idxA = listA.findIndex((l) => l.id === lesson.id);
  if (attemptNo > 2) {
    content().innerHTML = `${pageNavHtml({ backLabel: 'Back', showNext: false, center: 'Closed' })}
      <h2 class="page-title">${escapeHtml(lesson.title)}</h2>
      <div class="content-block"><span class="badge-soft badge-warn">Not Yet Competent</span><p class="mt-2 mb-0">Max 2 attempts used.</p></div>`;
    wirePageNav(() => renderCourseDetail(courseId), null); return;
  }
  const isRemediation = attemptNo === 2;
  content().innerHTML = `${pageNavHtml({ backLabel: 'Back', nextLabel: 'Next', showNext: idxA >= 0 && idxA < listA.length - 1, center: 'Attempt ' + attemptNo + '/2' })}
    <h2 class="page-title">${escapeHtml(lesson.title)}</h2>
    <p class="text-secondary small mb-2"><strong>Attempt ${attemptNo} of 2</strong>${isRemediation ? ' · Remediation' : ''}</p>
    <div class="content-block mb-3">${formatBody(lesson.body)}</div>
    <div class="content-block mb-3 small"><p class="mb-2"><strong>Instructions for upload</strong></p>
      <ul class="mb-2"><li>Put your <strong>full name</strong> at the end of the file name.</li></ul>
      <p class="mb-0"><strong>Important:</strong> Altering the assessment = immediate Not Yet Competent.</p></div>
    <label class="form-label">Written response</label>
    <textarea class="form-control mb-3" id="assignment-text" rows="5">${previous?.text ? escapeHtml(previous.text) : ''}</textarea>
    <div class="content-block mb-3"><p class="mb-2"><strong><i class="bi bi-upload"></i> Upload submissions</strong></p>
      <label class="form-label">Knowledge evidence</label>
      <input type="file" id="asg-knowledge" class="form-control mb-2" accept=".pdf,.jpg,.jpeg,.png,image/*,application/pdf">
      <label class="form-label">Practical evidence</label>
      <input type="file" id="asg-practical" class="form-control mb-2" accept=".pdf,.jpg,.jpeg,.png,image/*,application/pdf">
      <p class="text-secondary small mb-0">At least one file required.</p></div>
    <button class="btn btn-primary w-100" id="btn-submit-asg"><i class="bi bi-upload me-1"></i>${isRemediation ? 'Add a new attempt' : 'Upload and submit attempt ' + attemptNo}</button>
    ${pageNavBottomHtml({ backLabel: 'Back', nextLabel: 'Next', showNext: idxA >= 0 && idxA < listA.length - 1 })}`;
  wirePageNav(() => renderCourseDetail(courseId), () => openAdjacentLesson(courseId, lesson.id, 1));
  document.getElementById('btn-submit-asg')?.addEventListener('click', async () => {
    const text = document.getElementById('assignment-text').value.trim();
    if (text.length < 20) { alert('Write at least 20 characters.'); return; }
    let knowledge = await readFilesFromInput(document.getElementById('asg-knowledge'), false);
    let practical = await readFilesFromInput(document.getElementById('asg-practical'), false);
    if (previous?.attachments) { if (!knowledge) knowledge = previous.attachments.knowledge || null; if (!practical) practical = previous.attachments.practical || null; }
    if (!knowledge && !practical) { alert('Upload at least one file.'); return; }
    const list = getSubmissions();
    if (previous && previous.status === 'submitted') {
      const sub = list.find((x) => x.id === previous.id);
      if (sub) { sub.text = text; sub.status = 'submitted'; sub.attachments = { knowledge, practical }; sub.submittedAt = new Date().toISOString(); sub.attemptNo = previous.attemptNo || attemptNo; }
    } else {
      list.push({ id: uid(), assignmentId: lesson.assignmentId, courseId, learnerId: s.id, text, status: 'submitted', submittedAt: new Date().toISOString(), score: null, feedback: '', attemptNo, outcome: null, attachments: { knowledge, practical } });
    }
    save(KEYS.submissions, list);
    alert('Submission uploaded for attempt ' + attemptNo + '.');
    openAssignmentLesson(courseId, lesson, false);
  });
}

function renderGrading() {
  const subs = getSubmissions().filter((s) => s.status === 'submitted');
  const users = load(KEYS.users, []);
  content().innerHTML = `${pageNavHtml({ backLabel: 'Dashboard', showNext: false, center: 'Grade' })}
    <h2 class="page-title">Submissions to grade</h2>` +
    (subs.length === 0 ? '<div class="empty">Nothing to grade</div>' : subs.map((sub) => {
      const learner = users.find((u) => u.id === sub.learnerId);
      return `<div class="list-card"><h3>${escapeHtml(learner?.name || 'Learner')}</h3><div class="content-block mb-2">${formatBody(sub.text)}</div>
      <input type="number" class="form-control mb-2" id="score-${sub.id}" min="0" max="100" value="70">
      <textarea class="form-control mb-2" id="fb-${sub.id}" rows="2" placeholder="Feedback"></textarea>
      <div class="d-flex gap-2"><button class="btn btn-sm btn-primary" data-grade="${sub.id}" data-outcome="competent">Competent</button>
      <button class="btn btn-sm btn-outline-light" data-grade="${sub.id}" data-outcome="nyc">Not Yet Competent</button></div></div>`;
    }).join(''));
  wirePageNav(() => { renderAdminHome(); setNavActive('home'); }, null);
  content().querySelectorAll('[data-grade]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const id = btn.getAttribute('data-grade');
      const list = getSubmissions();
      const sub = list.find((s) => s.id === id);
      if (!sub) return;
      sub.status = 'graded';
      sub.score = Number(document.getElementById('score-' + id)?.value) || 0;
      const outcome = btn.getAttribute('data-outcome');
      sub.passed = outcome === 'competent' ? true : outcome === 'nyc' ? false : sub.score >= 70;
      sub.outcome = sub.passed ? 'Competent' : 'Not Yet Competent';
      sub.feedback = document.getElementById('fb-' + id)?.value?.trim() || '';
      save(KEYS.submissions, list);
      renderGrading();
    });
  });
}

function renderPayments() {
  const s = getSession();
  const courses = getCourses();
  if (isStaff(s)) {
    const enrolments = getEnrolments().filter((e) => e.status === 'approved');
    const users = load(KEYS.users, []);
    content().innerHTML = `${pageNavHtml({ backLabel: 'Dashboard', showNext: false, center: 'Payments' })}
      <h2 class="page-title">Payments</h2>` +
      enrolments.map((e) => {
        const learner = users.find((u) => u.id === e.learnerId);
        const course = courses.find((c) => c.id === e.courseId);
        const pay = getEnrolmentPaymentStatus(e);
        return `<div class="list-card"><h3>${escapeHtml(learner?.name || 'Learner')}</h3>
          <p class="text-secondary small">${escapeHtml(course?.programme || '')} · ${formatMoney(course?.feeZAR)}</p>
          <span class="badge-soft ${pay === 'paid' ? 'badge-ok' : 'badge-warn'}">${pay}</span>
          ${pay !== 'paid' && isAdmin(s) ? `<button class="btn btn-sm btn-primary mt-2" data-mark-paid="${e.id}">Mark as paid</button>` : ''}</div>`;
      }).join('') || '<div class="empty">No active enrolments</div>';
    wirePageNav(() => { renderAdminHome(); setNavActive('home'); }, null);
    content().querySelectorAll('[data-mark-paid]').forEach((btn) => {
      btn.addEventListener('click', () => {
        const row = getEnrolments().find((x) => x.id === btn.getAttribute('data-mark-paid'));
        if (row) recordPayment({ learnerId: row.learnerId, courseId: row.courseId, amount: getCourse(row.courseId)?.feeZAR || 0, method: 'manual' });
        renderPayments();
      });
    });
    return;
  }
  const allMine = getEnrolments().filter((e) => e.learnerId === s.id);
  const approved = allMine.filter((e) => e.status === 'approved');
  const pending = allMine.filter((e) => e.status === 'pending');
  const enrolments = approved.length ? [approved[0]] : [];
  let emptyHtml = '';
  if (!enrolments.length) {
    emptyHtml = pending.length
      ? `<div class="content-block mb-3"><p class="mb-2"><strong>Application pending approval</strong></p><p class="text-secondary small mb-0">Once approved, your fee appears here.</p></div><button class="btn btn-outline-light w-100" id="btn-pay-courses">View courses</button>`
      : `<div class="content-block mb-3"><p class="mb-2"><strong>No active programme</strong></p><p class="text-secondary small mb-0">Apply under Courses first.</p></div><button class="btn btn-primary w-100" id="btn-pay-courses">Go to Courses to apply</button>`;
  }
  content().innerHTML = `${pageNavHtml({ backLabel: 'Home', showNext: false, center: 'Pay' })}
    <h2 class="page-title">Fees & payments</h2><p class="page-subtitle">One programme fee at a time.</p>` +
    (enrolments.length === 0 ? emptyHtml : enrolments.map((e) => {
      const course = courses.find((c) => c.id === e.courseId);
      const pay = getEnrolmentPaymentStatus(e);
      return `<div class="list-card"><h3>${escapeHtml(course?.programme || course?.title || '')}</h3>
        <p class="mb-2">${formatMoney(course?.feeZAR)}</p>
        <span class="badge-soft ${pay === 'paid' ? 'badge-ok' : 'badge-warn'}">${pay === 'paid' ? 'Paid' : 'Outstanding — pay to unlock modules'}</span>
        ${pay !== 'paid' ? `<button class="btn btn-primary w-100 mt-3" data-pay="${e.courseId}">Pay with card (demo)</button>` : '<p class="text-secondary small mt-2 mb-0">Modules unlocked.</p>'}</div>`;
    }).join(''));
  wirePageNav(() => { renderLearnerHome(); setNavActive('home'); }, null);
  document.getElementById('btn-pay-courses')?.addEventListener('click', () => { renderCatalog(); setNavActive('courses'); });
  content().querySelectorAll('[data-pay]').forEach((btn) => btn.addEventListener('click', () => showCheckout(btn.getAttribute('data-pay'))));
}

function showCheckout(courseId) {
  const course = getCourse(courseId);
  const s = getSession();
  const active = getApprovedEnrolments(s.id)[0];
  if (!active || active.courseId !== courseId) { alert('Pay only for your one active programme.'); renderPayments(); return; }
  if (getEnrolmentPaymentStatus(active) === 'paid') { alert('Already paid.'); renderPayments(); return; }
  content().innerHTML = `${pageNavHtml({ backLabel: 'Payments', showNext: false, center: 'Checkout' })}
    <h2 class="page-title">Checkout</h2>
    <div class="content-block mb-3"><p class="mb-1"><strong>${escapeHtml(course?.programme || '')}</strong></p><p class="mb-0">Amount: <strong>${formatMoney(course?.feeZAR)}</strong></p></div>
    <button class="btn btn-primary w-100" id="btn-pay-now">Pay ${formatMoney(course?.feeZAR)}</button>`;
  wirePageNav(renderPayments, null);
  document.getElementById('btn-pay-now')?.addEventListener('click', () => {
    recordPayment({ learnerId: s.id, courseId, amount: course?.feeZAR || 0, method: 'card' });
    alert('Payment successful. Modules unlocked.');
    renderPayments();
  });
}

async function bootEduFlow() {
  await seedIfNeeded();
  enforceOneProgrammeRule();
  wireThemeToggles();
  setTimeout(() => document.getElementById('splash')?.classList.add('hide'), 400);
  document.getElementById('login-form')?.addEventListener('submit', (e) => {
    e.preventDefault();
    const result = login(document.getElementById('email').value.trim(), document.getElementById('password').value);
    const err = document.getElementById('login-error');
    if (!result.ok) { err.textContent = result.message; err.classList.remove('d-none'); return; }
    err.classList.add('d-none');
    if (result.needsVerify) showVerifyPanel(result.user);
    else if (result.needsApplication) showApplicationStep(result.user);
    else if (result.pending) showAwaitingApproval();
    else showApp();
  });
  document.getElementById('btn-register')?.addEventListener('click', (e) => { e.preventDefault(); showRegisterPanel(); });
  document.getElementById('btn-show-login')?.addEventListener('click', (e) => { e.preventDefault(); logout(); });
  document.getElementById('btn-show-login-from-reg')?.addEventListener('click', (e) => { e.preventDefault(); showLogin(); });
  document.getElementById('btn-show-login-from-verify')?.addEventListener('click', (e) => { e.preventDefault(); logout(); });
  document.getElementById('register-form')?.addEventListener('submit', async (e) => {
    e.preventDefault();
    const result = createLearnerAccount({ name: document.getElementById('reg-name')?.value.trim() || '', email: document.getElementById('reg-email')?.value.trim() || '', phone: document.getElementById('reg-phone')?.value.trim() || '', password: document.getElementById('reg-password')?.value || '', password2: document.getElementById('reg-password2')?.value || '' });
    const err = document.getElementById('reg-error');
    if (!result.ok) { if (err) { err.textContent = result.message; err.classList.remove('d-none'); } return; }
    await sendWelcomeEmail({ to: result.user.email, name: result.user.name, code: result.code });
    showVerifyPanel(result.user);
  });
  document.getElementById('verify-form')?.addEventListener('submit', (e) => {
    e.preventDefault();
    const s = getSession(); if (!s) return showLogin();
    const result = verifyLearnerEmail(s.id, document.getElementById('verify-code')?.value.trim() || '');
    const err = document.getElementById('verify-error');
    if (!result.ok) { if (err) { err.textContent = result.message; err.classList.remove('d-none'); } return; }
    showApplicationStep(result.user);
  });
  document.getElementById('btn-resend-code')?.addEventListener('click', async (e) => {
    e.preventDefault();
    const s = getSession(); if (!s) return;
    const users = load(KEYS.users, []);
    const user = users.find((u) => u.id === s.id);
    if (!user) return;
    user.verifyCode = generateVerifyCode();
    save(KEYS.users, users); setSession(user);
    await sendWelcomeEmail({ to: user.email, name: user.name, code: user.verifyCode });
    showVerifyPanel(user);
  });
  document.getElementById('apply-form')?.addEventListener('submit', async (e) => {
    e.preventDefault();
    const s = getSession(); if (!s) return showRegisterPanel();
    const hasDisability = !!document.getElementById('apply-disability')?.checked;
    const idCopy = await readFilesFromInput(document.getElementById('apply-id'), false);
    const matric = await readFilesFromInput(document.getElementById('apply-matric'), false);
    const otherCerts = await readFilesFromInput(document.getElementById('apply-other'), true);
    const disabilityNote = hasDisability ? await readFilesFromInput(document.getElementById('apply-note'), false) : null;
    const result = submitApplicationForUser(s.id, { programmeId: document.getElementById('apply-programme')?.value || '', notes: document.getElementById('apply-notes')?.value.trim() || '', documents: { hasDisability, idCopy, matric, otherCerts, disabilityNote } });
    const err = document.getElementById('apply-error');
    if (!result.ok) { if (err) { err.textContent = result.message; err.classList.remove('d-none'); } return; }
    showAwaitingApproval();
  });
  document.getElementById('btn-logout')?.addEventListener('click', logout);
  if (getSession()) showApp(); else showLogin();
}
if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', bootEduFlow);
else bootEduFlow();
