function renderCatalog() {
  const s = getSession();
  const courses = getCourses();
  const enrolments = getEnrolments();
  content().innerHTML = `<h2 class="page-title">${s.role === 'trainer' ? 'Programmes' : 'Course catalogue'}</h2><p class="text-secondary mb-3">Learners may only have <strong>one active programme</strong> at a time.</p>` +
    courses.map((c) => {
      const enr = enrolments.find((e) => e.learnerId === s.id && e.courseId === c.id);
      const status = enr ? enr.status : null;
      return `<div class="list-card"><h3>${escapeHtml(c.title)}</h3><p class="text-secondary small mb-2">${escapeHtml(c.description || '')}</p><div class="meta mb-2"><span class="badge-soft">NQF ${c.nqfLevel}</span><span class="badge-soft">${c.credits} credits</span></div>${s.role === 'trainer' || status === 'approved' ? `<button class="btn btn-sm btn-primary" data-open="${c.id}">Open modules</button>` : status === 'pending' ? `<span class="badge-soft badge-warn">Application pending</span>` : (s.role === 'learner' && hasActiveProgramme(s.id) ? `<span class="badge-soft">One programme at a time — finish current first</span>` : `<button class="btn btn-sm btn-primary" data-apply="${c.id}">Apply for this programme</button>`)}</div>`;
    }).join('');
  content().querySelectorAll('[data-open]').forEach((btn) => { btn.addEventListener('click', () => renderCourseDetail(btn.getAttribute('data-open'))); });
  content().querySelectorAll('[data-apply]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const courseId = btn.getAttribute('data-apply');
      if (hasActiveProgramme(s.id)) { alert('You already have an active programme. Complete it before applying for another.'); return; }
      if (getEnrolments().some((e) => e.learnerId === s.id && e.status === 'pending' && e.courseId !== courseId)) {
        alert('You already have a pending application. Wait for approval first.'); return;
      }
      const list = getEnrolments().filter((e) => !(e.learnerId === s.id && e.courseId === courseId));
      list.push({ id: uid(), courseId, learnerId: s.id, status: 'pending', enrolledAt: new Date().toISOString() });
      saveEnrolments(list);
      alert('Application submitted. One programme at a time.');
      renderCatalog();
    });
  });
}
function renderCourseDetail(courseId) {
  const c = getCourse(courseId);
  if (!c) return;
  const s = getSession();
  content().innerHTML = `<button class="btn btn-sm btn-outline-light mb-2" id="btn-back-course"><i class="bi bi-arrow-left"></i> Back</button><h2 class="page-title">${escapeHtml(c.title)}</h2><p class="text-secondary small">NQF ${c.nqfLevel} · ${c.credits} credits · SAQA ${c.saqaId}</p><h3 class="h6 text-secondary">Modules — tap to open</h3>${(c.lessons || []).map((l, i) => `<div class="lesson-card" data-lesson="${l.id}" role="button"><div class="num">${i + 1}</div><div style="flex:1"><h4>${escapeHtml(l.title)}</h4><p>${l.type}${l.type === 'assignment' ? ' · upload evidence' : ''} · <span style="color:#93c5fd">Open →</span></p></div></div>`).join('')}`;
  document.getElementById('btn-back-course')?.addEventListener('click', () => {
    if (s.role === 'trainer') { renderCatalog(); setNavActive('catalog'); }
    else { renderLearnerHome(); setNavActive('home'); }
  });
  content().querySelectorAll('[data-lesson]').forEach((el) => {
    el.addEventListener('click', () => {
      const lesson = (c.lessons || []).find((x) => x.id === el.getAttribute('data-lesson'));
      if (lesson) openLesson(courseId, lesson, s.role === 'trainer');
    });
  });
}
function openLesson(courseId, lesson, isTrainer) {
  if (lesson.type === 'quiz') {
    const quiz = lesson.quiz || { questions: [], passMark: 70 };
    const qs = quiz.questions || [];
    content().innerHTML = `<button class="btn btn-sm btn-outline-light mb-2" id="btn-back-lesson"><i class="bi bi-arrow-left"></i> Back</button><h2 class="page-title">${escapeHtml(lesson.title)}</h2><p class="text-secondary mb-3">${qs.length} questions · pass mark ${quiz.passMark}%</p>` +
      qs.map((q, qi) => `<div class="content-block mb-2"><p class="mb-2"><strong>${qi + 1}. ${escapeHtml(q.q)}</strong></p>${(q.options || []).map((opt, oi) => `<label class="d-block mb-1"><input type="radio" name="q${qi}" value="${oi}"> ${escapeHtml(opt)}</label>`).join('')}</div>`).join('') +
      `<div id="quiz-result" class="d-none content-block mb-2"></div><button class="btn btn-primary w-100" id="btn-submit-quiz">Submit answers</button>`;
    document.getElementById('btn-back-lesson')?.addEventListener('click', () => renderCourseDetail(courseId));
    document.getElementById('btn-submit-quiz')?.addEventListener('click', () => {
      let correct = 0;
      qs.forEach((q, qi) => {
        const sel = document.querySelector(`input[name="q${qi}"]:checked`);
        if (sel && Number(sel.value) === Number(q.correctIndex)) correct++;
      });
      const score = qs.length ? Math.round((correct / qs.length) * 100) : 0;
      const passed = score >= (quiz.passMark || 70);
      const attempts = load(KEYS.attempts, []);
      attempts.push({ id: uid(), learnerId: getSession().id, courseId, lessonId: lesson.id, score, passed, at: new Date().toISOString() });
      save(KEYS.attempts, attempts);
      const box = document.getElementById('quiz-result');
      if (box) {
        box.classList.remove('d-none');
        box.innerHTML = `<p class="mb-0"><span class="badge-soft ${passed ? 'badge-ok' : 'badge-warn'}">${passed ? 'Passed' : 'Not yet'} — ${score}%</span> (${correct}/${qs.length} correct)</p>`;
      }
      const btn = document.getElementById('btn-submit-quiz');
      if (btn) { btn.disabled = true; btn.textContent = 'Submitted'; }
    });
    return;
  }
  if (lesson.type === 'assignment') return openAssignmentLesson(courseId, lesson, isTrainer);
  content().innerHTML = `<button class="btn btn-sm btn-outline-light mb-2" id="btn-back-lesson"><i class="bi bi-arrow-left"></i> Back</button><h2 class="page-title">${escapeHtml(lesson.title)}</h2><p class="text-secondary small mb-3">${lesson.type}</p><div class="content-block">${formatBody(lesson.body)}</div>`;
  document.getElementById('btn-back-lesson')?.addEventListener('click', () => renderCourseDetail(courseId));
}
function openAssignmentLesson(courseId, lesson, isTrainer) {
  const s = getSession();
  const mine = getSubmissions().filter((x) => x.learnerId === s.id && x.assignmentId === lesson.assignmentId).pop();
  if (isTrainer) {
    content().innerHTML = `<button class="btn btn-sm btn-outline-light mb-2" id="btn-back-lesson"><i class="bi bi-arrow-left"></i> Back</button><h2 class="page-title">${escapeHtml(lesson.title)}</h2><div class="content-block">${formatBody(lesson.body)}</div>`;
    document.getElementById('btn-back-lesson')?.addEventListener('click', () => renderCourseDetail(courseId));
    return;
  }
  if (mine && mine.status === 'graded') {
    content().innerHTML = `<button class="btn btn-sm btn-outline-light mb-2" id="btn-back-lesson"><i class="bi bi-arrow-left"></i> Back</button><h2 class="page-title">${escapeHtml(lesson.title)}</h2><p><span class="badge-soft ${mine.passed ? 'badge-ok' : 'badge-warn'}">Score ${mine.score}%</span></p>${mine.feedback ? `<div class="content-block mb-2"><p class="mb-0"><strong>Feedback:</strong> ${escapeHtml(mine.feedback)}</p></div>` : ''}<div class="content-block mb-2" style="font-size:.9rem">${formatBody(mine.text)}</div><div class="content-block" style="font-size:.85rem"><p class="mb-1">Knowledge: ${fileLinkHtml(mine.attachments?.knowledge)}</p><p class="mb-0">Practical: ${fileLinkHtml(mine.attachments?.practical)}</p></div>${!mine.passed ? `<button class="btn btn-primary w-100" id="btn-resubmit">Resubmit</button>` : ''}`;
    document.getElementById('btn-back-lesson')?.addEventListener('click', () => renderCourseDetail(courseId));
    document.getElementById('btn-resubmit')?.addEventListener('click', () => showAssignmentSubmitForm(courseId, lesson, mine));
    return;
  }
  if (mine && mine.status === 'submitted') {
    content().innerHTML = `<button class="btn btn-sm btn-outline-light mb-2" id="btn-back-lesson"><i class="bi bi-arrow-left"></i> Back</button><h2 class="page-title">${escapeHtml(lesson.title)}</h2><div class="content-block mb-2"><p class="mb-0">Submitted — awaiting result.</p></div><div class="content-block" style="font-size:.85rem"><p class="mb-1">Knowledge: ${fileLinkHtml(mine.attachments?.knowledge)}</p><p class="mb-0">Practical: ${fileLinkHtml(mine.attachments?.practical)}</p></div>`;
    document.getElementById('btn-back-lesson')?.addEventListener('click', () => renderCourseDetail(courseId));
    return;
  }
  showAssignmentSubmitForm(courseId, lesson, null);
}
function showAssignmentSubmitForm(courseId, lesson, previous) {
  content().innerHTML = `<button class="btn btn-sm btn-outline-light mb-2" id="btn-back-lesson"><i class="bi bi-arrow-left"></i> Back</button><h2 class="page-title">${escapeHtml(lesson.title)}</h2><div class="content-block mb-3">${formatBody(lesson.body)}</div><label class="form-label">Knowledge response</label><textarea class="form-control mb-3" id="assignment-text" rows="5"></textarea><label class="form-label">Knowledge evidence</label><input type="file" id="asg-knowledge" class="form-control mb-2" accept=".pdf,.jpg,.jpeg,.png,image/*,application/pdf"><label class="form-label">Practical evidence</label><input type="file" id="asg-practical" class="form-control mb-3" accept=".pdf,.jpg,.jpeg,.png,.mp4,image/*,video/*,application/pdf"><div id="asg-error" class="alert alert-danger d-none py-2"></div><button class="btn btn-primary w-100" id="btn-submit-asg">Upload and submit</button>`;
  document.getElementById('btn-back-lesson')?.addEventListener('click', () => renderCourseDetail(courseId));
  document.getElementById('btn-submit-asg')?.addEventListener('click', async () => {
    const err = document.getElementById('asg-error');
    err?.classList.add('d-none');
    const text = document.getElementById('assignment-text').value.trim();
    if (text.length < 20) { if (err) { err.textContent = 'Write at least 20 characters.'; err.classList.remove('d-none'); } return; }
    const btn = document.getElementById('btn-submit-asg');
    if (btn) { btn.disabled = true; btn.textContent = 'Uploading…'; }
    try {
      const knowledge = await readFilesFromInput(document.getElementById('asg-knowledge'), false);
      const practical = await readFilesFromInput(document.getElementById('asg-practical'), false);
      if (!knowledge && !practical) { if (err) { err.textContent = 'Upload at least one file.'; err.classList.remove('d-none'); } if (btn) { btn.disabled = false; btn.textContent = 'Upload and submit'; } return; }
      const s = getSession();
      const list = getSubmissions();
      if (previous) {
        const sub = list.find((x) => x.id === previous.id);
        if (sub) { sub.text = text; sub.status = 'submitted'; sub.attachments = { knowledge, practical }; sub.submittedAt = new Date().toISOString(); }
      } else {
        list.push({ id: uid(), assignmentId: lesson.assignmentId, courseId, learnerId: s.id, text, status: 'submitted', submittedAt: new Date().toISOString(), score: null, feedback: '', attachments: { knowledge, practical } });
      }
      save(KEYS.submissions, list);
      alert('Submitted.');
      openAssignmentLesson(courseId, lesson, false);
    } catch (e) {
      if (err) { err.textContent = 'Upload failed.'; err.classList.remove('d-none'); }
      if (btn) { btn.disabled = false; btn.textContent = 'Upload and submit'; }
    }
  });
}
function renderGrading() {
  const subs = getSubmissions().filter((s) => s.status === 'submitted');
  const users = load(KEYS.users, []);
  content().innerHTML = `<h2 class="page-title">To grade</h2>` +
    (subs.length === 0 ? '<div class="empty">Nothing to grade</div>' : subs.map((sub) => {
      const learner = users.find((u) => u.id === sub.learnerId);
      return `<div class="list-card"><h3>${escapeHtml(learner?.name || 'Learner')}</h3><div class="content-block mb-2" style="font-size:.9rem">${formatBody(sub.text)}</div><div class="content-block mb-2" style="font-size:.85rem"><p class="mb-1">Knowledge: ${fileLinkHtml(sub.attachments?.knowledge)}</p><p class="mb-0">Practical: ${fileLinkHtml(sub.attachments?.practical)}</p></div><label class="form-label small">Score %</label><input type="number" class="form-control mb-2" id="score-${sub.id}" min="0" max="100" value="70"><label class="form-label small">Feedback to learner</label><textarea class="form-control mb-2" id="fb-${sub.id}" rows="2" placeholder="What went well / what to improve"></textarea><button class="btn btn-sm btn-primary" data-grade="${sub.id}">Save grade</button></div>`;
    }).join(''));
  content().querySelectorAll('[data-grade]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const id = btn.getAttribute('data-grade');
      const list = getSubmissions();
      const sub = list.find((s) => s.id === id);
      if (!sub) return;
      sub.status = 'graded';
      sub.score = Number(document.getElementById('score-' + id)?.value) || 0;
      sub.passed = sub.score >= 70;
      sub.feedback = document.getElementById('fb-' + id)?.value?.trim() || '';
      save(KEYS.submissions, list);
      const learner = users.find((u) => u.id === sub.learnerId);
      if (learner) pushMessage(learner.id, learner.gmail || learner.email, 'Assignment graded — ' + sub.score + '%', (sub.passed ? 'Passed. ' : 'Not yet competent. ') + (sub.feedback || 'See your course for details.'));
      alert('Grade saved. Learner notified in Inbox.');
      renderGrading();
    });
  });
}
function renderAttendance() {
  const s = getSession();
  const courses = getCourses();
  const attendance = getAttendance();
  if (s.role === 'trainer') {
    const users = load(KEYS.users, []).filter((u) => u.role === 'learner');
    const rows = attendance.slice(0, 50);
    content().innerHTML = `<h2 class="page-title">Attendance register</h2><p class="text-secondary mb-3">Remote marks (one programme per learner).</p>${rows.length === 0 ? '<div class="empty">No attendance marked yet</div>' : rows.map((a) => { const learner = users.find((u) => u.id === a.learnerId); const course = courses.find((c) => c.id === a.courseId); return `<div class="list-card"><h3>${escapeHtml(learner?.name || 'Learner')}</h3><p class="text-secondary small mb-1">${escapeHtml(course?.title || a.courseId)}</p><p class="mb-0"><span class="badge-soft badge-ok">${escapeHtml(a.date)}</span> <span class="text-secondary small">${new Date(a.markedAt).toLocaleTimeString()}</span></p></div>`; }).join('')}`;
    return;
  }
  const enrolments = getApprovedEnrolments(s.id);
  const day = todayISODate();
  content().innerHTML = `<h2 class="page-title">Mark attendance</h2><p class="text-secondary mb-3">Today <strong>${day}</strong> · one active programme only.</p>${enrolments.length === 0 ? '<div class="empty">No active programme. Apply under Courses first.</div>' : enrolments.map((e) => { const c = courses.find((x) => x.id === e.courseId); const already = attendance.find((a) => a.learnerId === s.id && a.courseId === e.courseId && a.date === day); return `<div class="list-card"><h3>${escapeHtml(c?.title || 'Programme')}</h3>${already ? `<p class="mb-0"><span class="badge-soft badge-ok">Present today</span> <span class="text-secondary small">${new Date(already.markedAt).toLocaleTimeString()}</span></p>` : `<button class="btn btn-sm btn-primary" data-mark="${e.courseId}">Mark present (remote)</button>`}</div>`; }).join('')}`;
  content().querySelectorAll('[data-mark]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const result = markAttendance(s.id, btn.getAttribute('data-mark'), 'Remote session');
      if (!result.ok) alert(result.message);
      else renderAttendance();
    });
  });
}

async function bootEduFlow() {
  await seedIfNeeded();
  enforceOneProgrammeRule();
  setTimeout(() => document.getElementById('splash')?.classList.add('hide'), 400);
  document.getElementById('login-form')?.addEventListener('submit', (e) => {
    e.preventDefault();
    const result = login(document.getElementById('email').value.trim(), document.getElementById('password').value);
    const err = document.getElementById('login-error');
    if (!result.ok) { err.textContent = result.message; err.classList.remove('d-none'); return; }
    err.classList.add('d-none');
    if (result.needsApplication) showApplicationStep(result.user);
    else if (result.pending) showAwaitingApproval();
    else showApp();
  });
  document.getElementById('btn-register')?.addEventListener('click', (e) => { e.preventDefault(); showRegisterPanel(); });
  document.getElementById('btn-show-login')?.addEventListener('click', (e) => { e.preventDefault(); logout(); });
  document.getElementById('btn-show-login-from-reg')?.addEventListener('click', (e) => { e.preventDefault(); showLogin(); });
  document.getElementById('register-form')?.addEventListener('submit', (e) => {
    e.preventDefault();
    const err = document.getElementById('reg-error');
    err?.classList.add('d-none');
    const result = createLearnerAccount({ name: document.getElementById('reg-name')?.value.trim() || '', email: document.getElementById('reg-email')?.value.trim() || '', phone: document.getElementById('reg-phone')?.value.trim() || '', password: document.getElementById('reg-password')?.value || '', password2: document.getElementById('reg-password2')?.value || '' });
    if (!result.ok) { if (err) { err.textContent = result.message; err.classList.remove('d-none'); } else alert(result.message); return; }
    showApplicationStep(result.user);
  });
  document.getElementById('apply-form')?.addEventListener('submit', async (e) => {
    e.preventDefault();
    const err = document.getElementById('apply-error');
    err?.classList.add('d-none');
    const s = getSession();
    if (!s) { alert('Create an account first.'); showRegisterPanel(); return; }
    const submitBtn = e.target.querySelector('button[type="submit"]');
    if (submitBtn) { submitBtn.disabled = true; submitBtn.textContent = 'Uploading…'; }
    try {
      const hasDisability = !!document.getElementById('apply-disability')?.checked;
      const idCopy = await readFilesFromInput(document.getElementById('apply-id'), false);
      const matric = await readFilesFromInput(document.getElementById('apply-matric'), false);
      const otherCerts = await readFilesFromInput(document.getElementById('apply-other'), true);
      const disabilityNote = hasDisability ? await readFilesFromInput(document.getElementById('apply-note'), false) : null;
      const result = submitApplicationForUser(s.id, { programmeId: document.getElementById('apply-programme')?.value || '', notes: document.getElementById('apply-notes')?.value.trim() || '', documents: { hasDisability, idCopy, matric, otherCerts, disabilityNote } });
      if (!result.ok) { if (err) { err.textContent = result.message; err.classList.remove('d-none'); } if (submitBtn) { submitBtn.disabled = false; submitBtn.textContent = 'Submit application'; } return; }
      showAwaitingApproval();
    } catch (ex) {
      if (err) { err.textContent = 'Upload failed.'; err.classList.remove('d-none'); }
      if (submitBtn) { submitBtn.disabled = false; submitBtn.textContent = 'Submit application'; }
    }
  });
  document.getElementById('btn-logout')?.addEventListener('click', logout);
  if (getSession()) showApp(); else showLogin();
}
if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', bootEduFlow);
else bootEduFlow();
