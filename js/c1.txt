/** ========== Progress / certificates ========== */
function getProgressList() { return load(KEYS.progress, []); }
function saveProgressList(list) { save(KEYS.progress, list); }
function isLessonDone(learnerId, courseId, lessonId) {
  return getProgressList().some((p) => p.learnerId === learnerId && p.courseId === courseId && p.lessonId === lessonId && p.done);
}
function markLessonDone(learnerId, courseId, lessonId) {
  const list = getProgressList();
  const row = list.find((p) => p.learnerId === learnerId && p.courseId === courseId && p.lessonId === lessonId);
  if (row) { row.done = true; row.at = new Date().toISOString(); }
  else list.push({ id: uid(), learnerId, courseId, lessonId, done: true, at: new Date().toISOString() });
  saveProgressList(list);
}
function getCourseProgress(learnerId, courseId) {
  const c = getCourse(courseId);
  const lessons = c?.lessons || [];
  if (!lessons.length) return { done: 0, total: 0, pct: 0 };
  const done = lessons.filter((l) => isLessonDone(learnerId, courseId, l.id)).length;
  return { done, total: lessons.length, pct: Math.round((done / lessons.length) * 100) };
}
function getSubmissionsList() { return load(KEYS.submissions, []); }
function saveSubmissionsList(list) { save(KEYS.submissions, list); }
function ensureCertificate(learnerId, courseId) {
  const prog = getCourseProgress(learnerId, courseId);
  if (prog.pct < 100) return null;
  const list = getCertificates();
  let cert = list.find((x) => x.learnerId === learnerId && x.courseId === courseId);
  if (!cert) {
    const c = getCourse(courseId);
    cert = {
      id: uid(), learnerId, courseId,
      programme: c?.programme || c?.title || '',
      issuedAt: new Date().toISOString(),
      code: 'EF-' + Math.random().toString(36).slice(2, 8).toUpperCase(),
      status: 'eligible',
    };
    list.unshift(cert);
    save(KEYS.certificates, list);
  }
  return cert;
}

/** ========== Attendance (keep) ========== */
function markAttendance(learnerId, courseId) {
  const list = getAttendance();
  const day = todayISODate();
  const existing = list.find((a) => a.learnerId === learnerId && a.courseId === courseId && a.date === day);
  if (existing) return existing;
  const row = { id: uid(), learnerId, courseId, date: day, at: new Date().toISOString(), status: 'signed', signedName: null };
  list.unshift(row);
  save(KEYS.attendance, list);
  return row;
}
function signRegister(learnerId, courseId, signedName) {
  const list = getAttendance();
  const day = todayISODate();
  let row = list.find((a) => a.learnerId === learnerId && a.courseId === courseId && a.date === day);
  if (row) {
    if (row.status === 'approved') return { ok: false, message: 'Already approved for today.' };
    row.signedName = signedName || row.signedName;
    row.at = new Date().toISOString();
    row.status = 'signed';
  } else {
    row = { id: uid(), learnerId, courseId, date: day, at: new Date().toISOString(), status: 'signed', signedName: signedName || '' };
    list.unshift(row);
  }
  save(KEYS.attendance, list);
  return { ok: true, row };
}
function approveAttendance(attendanceId) {
  const list = getAttendance();
  const row = list.find((a) => a.id === attendanceId);
  if (!row) return { ok: false, message: 'Record not found.' };
  row.status = 'approved';
  row.approvedAt = new Date().toISOString();
  const s = getSession();
  row.approvedBy = s ? s.id : null;
  save(KEYS.attendance, list);
  return { ok: true, row };
}

      if (pending.length) {
        html += `<h3 class="h6 mt-2 mb-2">Awaiting approval (${pending.length})</h3>`;

  }
  const approved = getApprovedEnrolments(s.id);
  let html = pageNavBar({ backLabel: 'Back to Home' }) +
    `<h2 class="page-title">Attendance register</h2>
    <p class="page-subtitle">Sign for today · ${escapeHtml(day)}</p>
    <div class="content-block mb-3"><p class="small mb-0">Sign once per programme each day. Your facilitator approves the register.</p></div>`;
  if (!approved.length) {
    html += `<div class="empty">You need an <strong>approved</strong> programme before you can sign.<br><span class="text-secondary small">Apply under Courses, then wait for admin approval.</span></div>`;
    content().innerHTML = html;

    focusPageHeading();
    return;
  }

  approved.forEach((e) => {
    const c = courses.find((x) => x.id === e.courseId);

/** ========== Lessons: guide/content + quiz + assignment ========== */
function openLesson(courseId, lessonId, idx) {
  const s = getSession();
  const c = getCourse(courseId);
  const lessons = c?.lessons || [];
  const lesson = lessons.find((l) => l.id === lessonId) || lessons[idx];
  if (!lesson) return renderCourseDetail(courseId);
  const i = typeof idx === 'number' ? idx : lessons.findIndex((l) => l.id === lessonId);
  const hasPrev = i > 0;
  const hasNext = i < lessons.length - 1;
  const done = isLessonDone(s.id, courseId, lesson.id);

  if (lesson.type === 'quiz' && lesson.quiz) {
    const qz = lesson.quiz;
    let html = pageNavBar({ backLabel: 'Back to course', nextLabel: hasNext ? 'Next' : null }) +
      `<h2 class="page-title">${escapeHtml(lesson.title)}</h2>
      <p class="text-secondary small mb-2">Formative assessment · Pass mark ${qz.passMark || 70}% · ${done ? '<span class="badge-soft badge-ok">Completed</span>' : 'Not completed'}</p>
      <form id="quiz-form" class="content-block">`;
    (qz.questions || []).forEach((q, qi) => {
      html += `<div class="mb-3"><p class="mb-2"><strong>Q${qi + 1}.</strong> ${escapeHtml(q.q)}</p>`;
      (q.options || []).forEach((opt, oi) => {
        html += `<label class="d-block mb-1"><input type="radio" name="q${qi}" value="${oi}" class="me-2"> ${escapeHtml(opt)}</label>`;
      });
      html += `</div>`;
    });
    html += `<div id="quiz-result" class="mb-2"></div>
      <button type="submit" class="btn btn-primary w-100">Submit answers</button></form>`;
    content().innerHTML = html;
    wirePageNav(() => renderCourseDetail(courseId), hasNext ? () => openLesson(courseId, lessons[i + 1].id, i + 1) : null);
    document.getElementById('quiz-form')?.addEventListener('submit', (e) => {
      e.preventDefault();
      let correct = 0;
      const qs = qz.questions || [];
      qs.forEach((q, qi) => {
        const sel = document.querySelector(`input[name="q${qi}"]:checked`);
        if (sel && Number(sel.value) === Number(q.correctIndex)) correct += 1;
      });
      const pct = qs.length ? Math.round((correct / qs.length) * 100) : 0;
      const pass = pct >= (qz.passMark || 70);
      const result = document.getElementById('quiz-result');
      if (result) {
        result.innerHTML = pass
          ? `<div class="alert alert-success py-2">Score ${pct}% — Competent. Well done.</div>`
          : `<div class="alert alert-warning py-2">Score ${pct}% — Not yet competent. Review the module and try again.</div>`;
      }
      if (pass) {
        markLessonDone(s.id, courseId, lesson.id);
        ensureCertificate(s.id, courseId);
        announce('Assessment passed');
      } else announce('Not yet competent — retry after review');
    });
    focusPageHeading();
    return;
  }

  if (lesson.type === 'assignment') {
    const subs = getSubmissionsList().filter((x) => x.learnerId === s.id && x.courseId === courseId && x.lessonId === lesson.id);
    const latest = subs[0];
    let html = pageNavBar({ backLabel: 'Back to course', nextLabel: hasNext ? 'Next' : null }) +
      `<h2 class="page-title">${escapeHtml(lesson.title)}</h2>
      <p class="text-secondary small mb-2">Summative assignment · Attempt ${(latest?.attempt || 0) + (latest?.status === 'submitted' || latest?.status === 'graded' ? 0 : 1)} of 2</p>
      <div class="content-block mb-3">${formatBody(lesson.body || 'Complete the task and upload your file.')}</div>`;
    if (latest) {
      html += `<div class="list-card mb-3"><p class="mb-1"><strong>Last submission</strong> · ${escapeHtml(latest.status)}</p>
        <p class="small text-secondary mb-1">${latest.fileName ? escapeHtml(latest.fileName) : 'No file'} · ${latest.at ? new Date(latest.at).toLocaleString() : ''}</p>
        ${latest.grade ? `<p class="mb-0"><span class="badge-soft ${latest.grade === 'Competent' ? 'badge-ok' : 'badge-warn'}">${escapeHtml(latest.grade)}</span> ${escapeHtml(latest.feedback || '')}</p>` : '<p class="small mb-0 text-secondary">Awaiting facilitator grade</p>'}
      </div>`;
    }
    if (!latest || latest.grade === 'Not Yet Competent' || (latest.status === 'graded' && latest.attempt < 2 && latest.grade !== 'Competent')) {
      html += `<div class="content-block"><label class="form-label">Upload submission (PDF/image)</label>
        <input type="file" id="asg-file" class="form-control mb-2" accept=".pdf,.jpg,.jpeg,.png,image/*,application/pdf">
        <label class="form-label">Notes</label>
        <textarea id="asg-notes" class="form-control mb-2" rows="2" placeholder="Brief note for the facilitator"></textarea>
        <button type="button" class="btn btn-primary w-100" id="btn-submit-asg">Submit assignment</button></div>`;
    } else if (latest && latest.grade === 'Competent') {
      html += `<div class="content-block"><span class="badge-soft badge-ok">Competent</span> — This task is complete.</div>`;
    }
    content().innerHTML = html;
    wirePageNav(() => renderCourseDetail(courseId), hasNext ? () => openLesson(courseId, lessons[i + 1].id, i + 1) : null);
    document.getElementById('btn-submit-asg')?.addEventListener('click', async () => {
      const file = await readFilesFromInput(document.getElementById('asg-file'), false);
      const notes = document.getElementById('asg-notes')?.value || '';
      if (!file) { announce('Choose a file to upload'); return; }
      const attempt = (latest?.attempt || 0) + 1;
      if (attempt > 2) { announce('Maximum 2 attempts'); return; }
      const list = getSubmissionsList();
      list.unshift({
        id: uid(), learnerId: s.id, courseId, lessonId: lesson.id,
        assignmentId: lesson.assignmentId || lesson.id,
        status: 'submitted', attempt, notes,
        fileName: file.name, fileDataUrl: file.dataUrl, tooLarge: file.tooLarge,
        at: new Date().toISOString(), grade: null, feedback: '',
      });
      saveSubmissionsList(list);
      markLessonDone(s.id, courseId, lesson.id);
      announce('Assignment submitted');
      openLesson(courseId, lesson.id, i);
    });
    focusPageHeading();
    return;
  }

  // guide / content
  content().innerHTML = pageNavBar({ backLabel: 'Back to course', nextLabel: hasNext ? 'Next lesson' : null }) +
    `<h2 class="page-title">${escapeHtml(lesson.title)}</h2>
    <p class="text-secondary small mb-2">Lesson ${i + 1} of ${lessons.length} ${done ? '· <span class="badge-soft badge-ok">Done</span>' : ''}</p>
    <div class="content-block">${formatBody(lesson.body || 'Content for this lesson.')}</div>
    <div class="d-flex gap-2 mt-3">
      ${hasPrev ? '<button type="button" class="btn btn-outline-light" id="btn-prev-lesson"><i class="bi bi-arrow-left"></i> Previous</button>' : ''}
      <button type="button" class="btn btn-primary ${hasPrev ? 'ms-auto' : 'w-100'}" id="btn-mark-done">${done ? 'Completed · Continue' : 'Mark as Done'}</button>
    </div>`;
  wirePageNav(() => renderCourseDetail(courseId), hasNext ? () => openLesson(courseId, lessons[i + 1].id, i + 1) : null);
  document.getElementById('btn-prev-lesson')?.addEventListener('click', () => openLesson(courseId, lessons[i - 1].id, i - 1));
  document.getElementById('btn-mark-done')?.addEventListener('click', () => {
    markLessonDone(s.id, courseId, lesson.id);
    ensureCertificate(s.id, courseId);
    announce('Marked as done');
    if (hasNext) openLesson(courseId, lessons[i + 1].id, i + 1);
    else renderCourseDetail(courseId);
  });
  focusPageHeading();
}

/** ========== Grading trail (staff) ========== */
function renderGrading() {
  const s = getSession();
  if (!isStaff(s)) {
    content().innerHTML = pageNavBar({ backLabel: 'Back to Home' }) + `<h2 class="page-title">Grading</h2><div class="empty">Staff only.</div>`;
    wirePageNav(goHome);
    focusPageHeading();
    return;
  }
  const users = load(KEYS.users, []);
  const courses = getCourses();
  const list = getSubmissionsList().slice().sort((a, b) => String(b.at || '').localeCompare(String(a.at || '')));
  const pending = list.filter((x) => x.status === 'submitted');
  const graded = list.filter((x) => x.status === 'graded');
  let html = pageNavBar({ backLabel: 'Back to Home' }) +
    `<h2 class="page-title">Grading</h2>
    <p class="page-subtitle">Competent / Not yet competent · Max 2 attempts</p>
    <div class="content-block mb-3"><p class="small mb-0">Learners submit summative assignments from their modules. Grade each attempt and leave short feedback.</p></div>`;
  if (!list.length) {
    html += `<div class="empty">No submissions yet.<br><span class="text-secondary small">When a learner uploads an assignment, it appears here for grading.</span></div>`;
  } else {
    if (pending.length) {
      html += `<h3 class="h6 mb-2">To grade (${pending.length})</h3>`;
      pending.forEach((sub) => {
        const u = users.find((x) => x.id === sub.learnerId);
        const c = courses.find((x) => x.id === sub.courseId);
        html += `<div class="list-card">
          <h3>${escapeHtml(u?.name || 'Learner')}</h3>
          <p class="text-secondary small mb-1">${escapeHtml(c?.programme || '')} · Attempt ${sub.attempt || 1} · ${escapeHtml(sub.fileName || 'file')}</p>
          <p class="small mb-2">${escapeHtml(sub.notes || '')}</p>
          ${sub.fileDataUrl && !sub.tooLarge ? `<p class="small mb-2"><a href="${sub.fileDataUrl}" target="_blank" rel="noopener">Open file</a></p>` : ''}
          <label class="form-label">Feedback</label>
          <input type="text" class="form-control mb-2" id="fb-${sub.id}" placeholder="Short feedback">
          <div class="d-flex gap-2">
            <button type="button" class="btn btn-sm btn-primary" data-grade="${sub.id}" data-result="Competent">Competent</button>
            <button type="button" class="btn btn-sm btn-outline-light" data-grade="${sub.id}" data-result="Not Yet Competent">Not yet competent</button>
          </div></div>`;
      });
    }
    if (graded.length) {
      html += `<h3 class="h6 mt-3 mb-2">Recently graded</h3>`;
      graded.slice(0, 8).forEach((sub) => {
        const u = users.find((x) => x.id === sub.learnerId);
        html += `<div class="list-card"><h3>${escapeHtml(u?.name || 'Learner')} <span class="badge-soft ${sub.grade === 'Competent' ? 'badge-ok' : 'badge-warn'}">${escapeHtml(sub.grade || '')}</span></h3>
          <p class="small text-secondary mb-0">Attempt ${sub.attempt || 1} · ${escapeHtml(sub.feedback || '')}</p></div>`;
      });
    }
  }
  content().innerHTML = html;
  wirePageNav(goHome);
  content().querySelectorAll('[data-grade]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const id = btn.getAttribute('data-grade');
      const result = btn.getAttribute('data-result');
      const fb = document.getElementById('fb-' + id)?.value || '';
      const all = getSubmissionsList();
      const row = all.find((x) => x.id === id);
      if (!row) return;
      row.status = 'graded';
      row.grade = result;
      row.feedback = fb;
      row.gradedAt = new Date().toISOString();
      row.gradedBy = s.id;
      saveSubmissionsList(all);
      announce(result === 'Competent' ? 'Marked Competent' : 'Marked Not yet competent');
      renderGrading();
    });
  });
  focusPageHeading();
}

/** ========== Payments empty state ========== */
function renderPayments() {
  const s = getSession();
  const approved = getApprovedEnrolments(s.id);
  const courses = getCourses();
  const pays = getPayments().filter((p) => p.learnerId === s.id);
  let html = pageNavBar({ backLabel: 'Back to Home' }) +
    `<h2 class="page-title">Fees & payments</h2>
    <p class="page-subtitle">Pay once per active programme to unlock modules</p>`;
  if (!approved.length) {
    html += `<div class="empty">No approved programme to pay for.<br><span class="text-secondary small">After admin approval, your fee appears here.</span></div>`;
  } else {
    approved.forEach((e) => {
      const c = courses.find((x) => x.id === e.courseId);
      const pay = getEnrolmentPaymentStatus(e);
      html += `<div class="list-card"><h3>${escapeHtml(c?.programme || '')}</h3>
        <p class="mb-2">${formatMoney(c?.feeZAR)} · <span class="badge-soft ${pay === 'paid' ? 'badge-ok' : 'badge-warn'}">${pay === 'paid' ? 'Paid' : 'Payment outstanding'}</span></p>
        ${pay !== 'paid' ? `<button type="button" class="btn btn-sm btn-primary" data-pay="${e.courseId}">Pay now (demo)</button>` : '<p class="small text-secondary mb-0">Modules unlocked on Courses / Home.</p>'}
      </div>`;
    });
  }
  if (pays.length) {
    html += `<h3 class="h6 mt-3 mb-2">Payment history</h3>`;
    pays.slice(0, 5).forEach((p) => {
      html += `<div class="list-card"><p class="mb-0 small">${formatMoney(p.amount)} · ${escapeHtml(p.ref || '')} · ${p.at ? new Date(p.at).toLocaleDateString() : ''}</p></div>`;
    });
  }
  content().innerHTML = html;
  wirePageNav(goHome);
  content().querySelectorAll('[data-pay]').forEach((btn) => btn.addEventListener('click', () => {
    const courseId = btn.getAttribute('data-pay');
    const c = getCourse(courseId);
    recordPayment({ learnerId: s.id, courseId, amount: c?.feeZAR || 0, method: 'card' });
    renderPayments();
  }));
  focusPageHeading();
}

async function bootEduFlow() {
  try { await seedIfNeeded(); } catch (e) { console.error(e); }
  try { ensureDemoEnrolments(); } catch (e) { console.error(e); }
  if (typeof enforceOneProgrammeRule === 'function') enforceOneProgrammeRule();
  if (typeof wireThemeToggles === 'function') wireThemeToggles();
  setTimeout(() => document.getElementById('splash')?.classList.add('hide'), 400);
  document.getElementById('login-form')?.addEventListener('submit', (e) => {
    e.preventDefault();
    try { ensureDemoEnrolments(); } catch (_) {}
    const result = login(document.getElementById('email').value.trim(), document.getElementById('password').value);
    const err = document.getElementById('login-error');
    if (!result.ok) { if (err) { err.textContent = result.message; err.classList.remove('d-none'); } return; }
    if (err) err.classList.add('d-none');
    if (result.needsVerify) showVerifyPanel(result.user);
    else if (result.needsApplication) showApplicationStep(result.user);
    else if (result.pending) showAwaitingApproval();
    else showApp();
  });
  document.getElementById('btn-register')?.addEventListener('click', (e) => { e.preventDefault(); showRegisterPanel(); });
  document.getElementById('btn-show-login-from-reg')?.addEventListener('click', (e) => { e.preventDefault(); showLogin(); });
  document.getElementById('btn-logout')?.addEventListener('click', logout);
  document.getElementById('skip-to-content')?.addEventListener('click', (e) => {
    e.preventDefault();
    const main = document.getElementById('content');
    if (main) { main.setAttribute('tabindex', '-1'); main.focus(); announce('Main content'); }
  });
  document.getElementById('apply-form')?.addEventListener('submit', async (e) => {
    e.preventDefault();
    const s = getSession();
    if (!s) return;
    const err = document.getElementById('apply-error');
    try {
      const programmeId = document.getElementById('apply-programme')?.value;
      const idCopy = await readFilesFromInput(document.getElementById('apply-id'), false);
      const matric = await readFilesFromInput(document.getElementById('apply-matric'), false);
      const other = await readFilesFromInput(document.getElementById('apply-other'), true);
      const hasDisability = !!document.getElementById('apply-disability')?.checked;
      const disabilityNote = hasDisability ? await readFilesFromInput(document.getElementById('apply-note'), false) : null;
      const notes = document.getElementById('apply-notes')?.value || '';
      const result = submitApplicationForUser(s.id, { programmeId, notes, documents: { idCopy, matric, other, hasDisability, disabilityNote } });
      if (!result.ok) { if (err) { err.textContent = result.message; err.classList.remove('d-none'); } return; }
      showAwaitingApproval();
    } catch (ex) {
      if (err) { err.textContent = 'Could not read files.'; err.classList.remove('d-none'); }
    }
  });
  if (getSession()) showApp(); else showLogin();
}
if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', bootEduFlow);
else bootEduFlow();
